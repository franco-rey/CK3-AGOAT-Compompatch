# Observer Stack P33 -- birth effects for the canon children of Aegon IV
# Targaryen, IDs 98-104. AGOT+ references these from events .0882 / .0883 but
# never defines them, so every matching pregnancy was terminated and no child
# was created.
#
# Modelled on AGOT+'s own implemented siblings (e.g. Targaryen_94). Names,
# genders, cultures and houses follow AGOT's canonical history entries for
# Targaryen_98-104 in 00_agot_char_dragonstone_ancestors.txt and
# 00_agot_char_braavos_ancestors.txt. Appearance is copied from those same
# canon characters, which ship full DNA in AGOT's dna_data.
asoiaf_canon_children_Targaryen_98_birth_effect = { #Alysanne (bastard of Aegon IV)
	create_character = {
		age = 0
		name = "Alysanne"
		father = scope:father
		real_father = scope:real_father
		mother = scope:mother
		gender = female
		faith = scope:real_father.faith
		culture = scope:real_father.culture
		dynasty_house = scope:real_father.house
		location = scope:mother.location
		employer = scope:mother.employer
		random_traits = no
		save_scope_as = child
		after_creation = {
			asoiaf_canon_children_parentage_assignment_effect = yes
			asoiaf_canon_children_join_travel_plan_effect = yes
			set_global_variable = asoiaf_canon_children_Targaryen_98_born_variable
			make_trait_inactive = asoiaf_Targaryen_98_trait
			copy_inheritable_appearance_from = character:Targaryen_98
			add_character_flag = has_scripted_appearance
			add_character_flag = canon_status_canon
			learn_language = language_agot_westeros
		}
	}
	hidden_effect = {
		scope:mother = { trigger_event = birth.1001 } #regular
		scope:mother = { trigger_event = birth.1002 } #secret bastard
		scope:father = { trigger_event = birth.1003 } #regular
	}
}
