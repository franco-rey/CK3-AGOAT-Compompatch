# A sworn protector who can no longer perform the office should not silently
# occupy a slot forever. AI rulers retire such guards automatically; players
# receive a formal choice and may retain them despite their infirmity.
agot_guard_positions_incapacitated_guard_maintenance = {
	trigger = {
		agot_guard_positions_has_active_guard_trigger = yes
		any_councillor = {
			agot_guard_positions_is_active_guard_member_trigger = { COURT_OWNER = root }
			OR = { has_trait = incapable has_trait = infirm }
			NOT = { has_character_flag = agot_guard_positions_incapacity_reviewed }
		}
	}
	effect = {
		random_councillor = {
			limit = {
				agot_guard_positions_is_active_guard_member_trigger = { COURT_OWNER = root }
				OR = { has_trait = incapable has_trait = infirm }
				NOT = { has_character_flag = agot_guard_positions_incapacity_reviewed }
			}
			save_scope_as = incapacitated_guard
			add_character_flag = agot_guard_positions_incapacity_reviewed
		}
		if = {
			limit = { is_ai = yes }
			scope:incapacitated_guard = { add_character_flag = agot_guard_positions_honorable_retirement }
			agot_guard_positions_dismiss_custom_guard_member_effect = { GUARD = scope:incapacitated_guard }
			scope:incapacitated_guard = { remove_character_flag = agot_guard_positions_honorable_retirement }
			agot_guard_positions_repair_guard_assignments_effect = yes
		}
		else = { trigger_event = agot_guard_positions.3020 }
	}
}

# Keep native/history modifiers from stacking a second academy on an already
# enrolled student. This is intentionally court-local and quarterly rather than
# a world scan.
agot_guard_positions_training_exclusivity_maintenance = {
	trigger = {
		OR = {
			ruler_has_kingsguard_trigger = yes
			has_character_flag = has_battleguard
			has_character_flag = has_winged_knights
			has_character_flag = has_swords_of_braavos
			has_character_flag = has_rainbow_guard
			has_character_flag = has_kingsmen
			has_character_flag = has_queensguard
			agot_guard_positions_has_greenhand_trigger = yes
		}
	}
	effect = {
		# Taking another binding oath or becoming otherwise unavailable ends
		# academy enrollment even while the student's original order still exists.
		every_in_list = {
			variable = agot_guard_positions_trainee_list
			limit = {
				has_variable = agot_guard_positions_training_owner
				var:agot_guard_positions_training_owner ?= root
				OR = {
					is_imprisoned = yes
					is_ruler = yes
					is_married = yes
					has_trait = incapable
					has_trait = dragon
					has_trait = nightswatch
					has_trait = maester
					has_trait = devoted
					has_trait = order_member
					has_trait = septon
					agot_guard_positions_is_any_guard_member_trigger = yes
				}
			}
			agot_guard_positions_clear_training_ambition_effect = yes
		}
		every_in_list = {
			variable = agot_guard_positions_trainee_list
			limit = {
				agot_guard_positions_has_any_training_ambition_trigger = yes
				has_variable = agot_guard_positions_training_owner
				var:agot_guard_positions_training_owner ?= root
			}
			agot_guard_positions_enforce_single_training_order_effect = { OWNER = root }
		}
	}
}

agot_guard_positions_training_maintenance = {
	trigger = {
		OR = {
			ruler_has_kingsguard_trigger = yes
			has_character_flag = has_battleguard
			has_character_flag = has_winged_knights
			has_character_flag = has_rainbow_guard
			has_character_flag = has_kingsmen
			has_character_flag = has_queensguard
			agot_guard_positions_has_greenhand_trigger = yes
		}
	}
	effect = {
		# Native AGOT history modifiers require a discovery pass, but this belongs
		# to yearly academy maintenance rather than quarterly ward cleanup.
		agot_guard_positions_adopt_native_kingsguard_trainees_effect = yes
		agot_guard_positions_select_school_master_effect = yes
		agot_guard_positions_enroll_canon_minor_guards_effect = yes
		agot_guard_positions_seed_white_book_canon_trainees_effect = yes
		every_councillor = {
			limit = { has_variable = agot_guard_positions_training_owner agot_guard_positions_is_any_guard_member_trigger = yes }
			agot_guard_positions_clear_training_ambition_effect = yes
		}
		# Academy students cannot complete knighthood or Thaneship before their
		# academy graduation, even if an external ceremony makes them eligible.
		# Iterates the maintained per-owner trainee list rather than every living
		# character in the world; agot_guard_positions_trainee_list is kept in sync
		# with var:agot_guard_positions_training_owner at every enrollment/release
		# point (see agot_guard_positions_enroll_guard_trainee_effect and
		# agot_guard_positions_clear_training_ambition_effect). The owner-variable
		# check is kept as a defensive redundancy, not because it's load-bearing.
		every_in_list = {
			variable = agot_guard_positions_trainee_list
			limit = {
				agot_guard_positions_has_any_training_ambition_trigger = yes
				var:agot_guard_positions_training_owner ?= root
				OR = { is_agot_knight_trigger = yes has_trait = agot_thane }
				NOT = { agot_guard_positions_is_academy_graduate_trigger = yes }
			}
			agot_guard_positions_remove_premature_student_knighthood_effect = yes
		}
		every_in_list = {
			variable = agot_guard_positions_trainee_list
			limit = {
				agot_guard_positions_has_any_training_ambition_trigger = yes
				var:agot_guard_positions_training_owner ?= root
			}
			save_scope_as = guard_trainee
			agot_guard_positions_enforce_single_training_order_effect = { OWNER = root }
			agot_guard_positions_assign_regional_academy_house_effect = yes
			agot_guard_positions_update_training_rank_effect = yes
			agot_guard_positions_sync_academy_squire_path_effect = yes
			if = {
				limit = { age >= 25 NOT = { agot_guard_positions_is_academy_graduate_trigger = yes } }
				agot_guard_positions_become_academy_instructor_effect = yes
			}
			else_if = {
				limit = { NOT = { agot_guard_positions_is_training_for_active_order_trigger = { OWNER = root } } }
				agot_guard_positions_clear_training_ambition_effect = yes

# If Dragonstone passes away from the bookmark Rhaenyra before she wins the
# Iron Throne, her personal Queensguard does not pass to the new title holder.


# This is Rhaenyra's one historical household institution, not a Dragonstone
# order that can pass to an heir.  Her death permanently destroys it unless it
# has already been converted into the royal Kingsguard.








# ============================================================================
# Observer Stack P39 - AGOT: Royal Guards (3733779333) pending-event flood
#
# This on_action is registered on on_title_gain, on_title_gain_inheritance and
# on_title_gain_usurpation, and its trigger only asked "is this character a
# guard member". Every firing queues agot_guard_positions.1006, which fans out
# to SIX agot_guard_positions.1004 (days 1/3/7 on both employer and liege) plus
# one agot_guard_positions.1007 -- with no de-duplication.
#
# In the 8183 run that produced:
#     agot_guard_positions.1004   12,104 queued onto ONE character (85912)
#     agot_guard_positions.1007   14,116 queued onto THREE characters
# 26,220 events -- 72% of the entire pending-event queue -- with 12,696 of them
# landing on a single day. The daily tick could not drain the queue, so the
# clock stopped while the UI stayed responsive.
#
# The mod's author already designed the guard for this: agot_guard_positions.1004
# tests NOT = { has_character_flag = agot_guard_positions_inheritance_review_pending }.
# Nothing anywhere ever set that flag, so the check could never fire. It is set
# here, at the one place the review cycle starts, and the same flag now gates
# re-entry.
#
# Duration is 12 days: .1006 lands at day 1 and its own fan-out reaches day 11
# (1 + 10 for .1007), so one full review cycle completes inside the window. A
# guard can therefore have at most one review in flight instead of thousands.
# The repair sweep itself is unchanged and remains idempotent.
# ============================================================================
agot_guard_positions_guard_inherits_title = {
	trigger = {
		NOT = { has_character_flag = agot_guard_positions_inheritance_review_pending }
		OR = {
			agot_guard_positions_is_queensguard_trigger = yes
			agot_guard_positions_is_battleguard_trigger = yes
			agot_guard_positions_is_winged_knight_trigger = yes
			agot_guard_positions_is_rainbow_guard_trigger = yes
			agot_guard_positions_is_kingsmen_trigger = yes
			agot_guard_positions_is_greenhand_trigger = yes
		}
	}

	effect = {
		add_character_flag = blocked_from_leaving
		add_character_flag = { flag = agot_guard_positions_inherited_title_as_guard days = 30 }
		add_character_flag = { flag = agot_guard_positions_inheritance_review_pending days = 12 }
		trigger_event = { id = agot_guard_positions.1006 days = 1 }
	}
}

					has_council_position = greenhand_4

				}
			}

			agot_guard_positions_is_battleguard_trigger = yes
			agot_guard_positions_is_winged_knight_trigger = yes
			agot_guard_positions_is_rainbow_guard_trigger = yes
			agot_guard_positions_is_kingsmen_trigger = yes
			agot_guard_positions_is_greenhand_trigger = yes
		}
		exists = scope:old_employer
		scope:old_employer = {
			OR = {
				has_character_flag = has_rainbow_guard
				has_character_flag = has_battleguard
				has_character_flag = has_winged_knights
				has_character_flag = has_kingsmen
				has_character_flag = has_queensguard
				agot_guard_positions_has_greenhand_trigger = yes
			}
		}
		trigger_if = {
			limit = { exists = scope:new_employer }
			NOT = { scope:new_employer = scope:old_employer }
		}
	}

	effect = {
		add_character_flag = blocked_from_leaving
		scope:old_employer = {
			trigger_event = { id = agot_guard_positions.1004 days = 1 }
		}
		if = {
			limit = { is_landed = no }
			scope:old_employer = { add_courtier ?= root }
		}
	}
}

agot_guard_positions_guard_member_death = {
	trigger = {
		OR = {
			agot_guard_positions_is_queensguard_trigger = yes
			agot_guard_positions_is_battleguard_trigger = yes
			agot_guard_positions_is_winged_knight_trigger = yes
			agot_guard_positions_is_rainbow_guard_trigger = yes
			agot_guard_positions_is_kingsmen_trigger = yes
			agot_guard_positions_is_greenhand_trigger = yes
		}
		OR = {
			liege_or_court_owner = {
				OR = {
					has_character_flag = has_queensguard
					has_character_flag = has_battleguard
					has_character_flag = has_winged_knights
					has_character_flag = has_rainbow_guard
					has_character_flag = has_kingsmen
					agot_guard_positions_has_greenhand_trigger = yes
					has_title = title:e_the_north
					has_title = title:e_the_stormlands
					has_title = title:k_dragonstone
				}
			}
			top_liege = {
				OR = {
					has_character_flag = has_queensguard
					has_character_flag = has_battleguard
					has_character_flag = has_winged_knights
					has_character_flag = has_rainbow_guard
					has_character_flag = has_kingsmen
					agot_guard_positions_has_greenhand_trigger = yes
					has_title = title:e_the_north
					has_title = title:e_the_stormlands
					has_title = title:k_dragonstone
				}
			}
		}
	}

	effect = {
		save_scope_as = dead_guard
		save_scope_as = dead_kingsguard
		house.house_head ?= { save_scope_as = dead_guard_family_representative }
		if = { limit = { exists = liege_or_court_owner } agot_guard_positions_record_death_effect = { HOLDER = liege_or_court_owner } }
		if = {
			limit = { num_of_relation_agot_squire > 0 }
			if = {
				limit = { num_of_relation_agot_squire > 1 exists = liege_or_court_owner }
				# Earlier rounds are abstracted; the two strongest surviving pupils
				# contest the final with blunted weapons in the ruler's event.
				ordered_relation = {
					type = agot_squire
					position = 0
					order_by = prowess
					save_scope_as = memorial_squire_1
				}
				ordered_relation = {
					type = agot_squire
					position = 1
					order_by = prowess
					save_scope_as = memorial_squire_2
				}
				liege_or_court_owner = { trigger_event = agot_guard_legacy.0001 }
			}
			else = {
				random_relation = {
					type = agot_squire
					agot_guard_positions_create_squire_memorial_effect = { DEAD_GUARD = root }
				}
			}
		}
		if = {
			limit = {
				liege_or_court_owner = {
					OR = {
						has_character_flag = has_queensguard
						has_character_flag = has_battleguard
						has_character_flag = has_winged_knights
						has_character_flag = has_rainbow_guard
						has_character_flag = has_kingsmen
						agot_guard_positions_has_greenhand_trigger = yes
						has_title = title:e_the_north
						has_title = title:e_the_stormlands
						has_title = title:k_dragonstone
					}
				}
			}
			liege_or_court_owner = {
				agot_guard_positions_sync_guard_flags_from_titles_effect = yes
				save_scope_as = king
				trigger_event = agot_guard_legacy.0010
			}
		}
		else_if = {
			limit = {
				top_liege = {
					OR = {
						has_character_flag = has_queensguard
						has_character_flag = has_battleguard
						has_character_flag = has_winged_knights
						has_character_flag = has_rainbow_guard
						has_character_flag = has_kingsmen
						agot_guard_positions_has_greenhand_trigger = yes
						has_title = title:e_the_north
						has_title = title:e_the_stormlands
						has_title = title:k_dragonstone
					}
				}
			}
			top_liege = {
				agot_guard_positions_sync_guard_flags_from_titles_effect = yes
				save_scope_as = king
				trigger_event = agot_guard_legacy.0010
			}
		}
	}
}
