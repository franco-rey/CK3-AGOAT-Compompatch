accou_marriage_candidate_quality_value = {
	value = 0


	add = scope:suitor_opinion


	if = {
		limit = {
			scope:suitor = {
				has_relation_lover = prev
			}
		}
		add = {
			value = 30
			scope:suitor = {
				if = {
					limit = { has_relation_soulmate = prev }
					multiply = 2
				}
			}
		}
	}


	if = {
		limit = {
			scope:suitor = {
				is_attracted_to_gender_of = prev
			}
		}
		add = {
			value = attraction
			scope:suitor = {
				if = {
					limit = { has_trait = lustful }
					multiply = 2
				}
				else_if = {
					limit = { has_trait = chaste }
					multiply = 0.5
				}
			}
		}
	}


	if = {
		limit = {
			sex_opposite_of = scope:suitor
			scope:suitor = {
				accou_is_fertile_trigger = yes
			}
		}
		add = {
			if = {
				limit = { accou_is_fertile_trigger = yes }
				value = fertility
			}
			else = {
				value = 0
			}
			subtract = define:NCharacter|BASE_FERTILITY
			multiply = 50
		}
	}


	add = {
		value = health
		subtract = define:NCharacter|BASE_HEALTH
		add = 1
		multiply = 5
	}


	if = {
		limit = { has_std_trigger = yes }
		add = -50
	}


	add = {
		value = 36
		subtract = age
	}


	add = {
		if = {
			limit = { age < scope:suitor.age }
			value = age
			subtract = scope:suitor.age
			multiply = 0.5
		}
		else = {
			value = scope:suitor.age
			subtract = age
		}
	}


	add = {
		value = 0
		if = {
			limit = { exists = dynasty }
			add = dynasty.dynasty_prestige_level
		}
		if = {
			limit = { exists = scope:suitor.dynasty }
			subtract = scope:suitor.dynasty.dynasty_prestige_level
		}
		multiply = 5
		scope:suitor = {
			if = {
				limit = { has_trait = ambitious }
				multiply = 2
			}
			else_if = {
				limit = { has_trait = content }
				multiply = 0.5
			}
		}
	}

	save_temporary_scope_as = target_character


	if = {
		limit = {
			faith != scope:suitor.faith
		}
		add = {
			value = scope:suitor.accou_faith_hostility_opinion_value
			multiply = 0.5
			scope:suitor = {
				if = {
					limit = { has_trait = zealous }
					multiply = 2
				}
				else_if = {
					limit = { has_trait = cynical }
					multiply = 0.5
				}
			}
		}
	}


	if = {
		limit = {
			culture != scope:suitor.culture
		}
		add = {
			value = scope:suitor.accou_cultural_acceptance_opinion_value
			multiply = 0.5
			scope:suitor = {
				if = {
					limit = { has_trait = gregarious }
					multiply = 2
				}
				else_if = {
					limit = { has_trait = shy }
					multiply = 0.5
				}
			}
		}
	}


	add = {
		value = accou_spouse_personality_value
		multiply = 10
	}


	if = {
		limit = {
			sex_opposite_of = scope:suitor
			accou_is_fertile_trigger = yes
			scope:suitor = {
				accou_is_fertile_trigger = yes
			}
		}
		add = {
			value = genetic_character_value
			multiply = 10
		}
	}


	add = {
		value = sum_of_all_skills_value
		subtract = 40
		multiply = 2
	}
	add = {
		value = prowess
		subtract = 8
	}

	round = yes
}

accou_faith_hostility_opinion_value = {

	if = {
		limit = {
			faith = {
				faith_hostility_level = {
					target = scope:target_character.faith
					value < faith_astray_level
				}
			}
		}
		value = 0
	}
	else_if = {
		limit = {
			faith = {
				faith_hostility_level = {
					target = scope:target_character.faith
					value < faith_hostile_level
				}
			}
		}
		value = -10
	}
	else_if = {
		limit = {
			faith = {
				faith_hostility_level = {
					target = scope:target_character.faith
					value < faith_evil_level
				}
			}
		}
		value = -20
	}
	else = {
		value = -30
	}
}

accou_cultural_acceptance_opinion_value = {
	value = define:NCharacterOpinion|ZERO_CULTURE_ACCEPTANCE
	if = {
		limit = {
			culture = {
				cultural_acceptance = {
					target = scope:target_character.culture
					value = 100
				}
			}
		}
		multiply = 0
	}
	else_if = {
		limit = {
			culture = {
				cultural_acceptance = {
					target = scope:target_character.culture
					value >= 75
				}
			}
		}
		multiply = 0.25
	}
	else_if = {
		limit = {
			culture = {
				cultural_acceptance = {
					target = scope:target_character.culture
					value >= 50
				}
			}
		}
		multiply = 0.5
	}
	else_if = {
		limit = {
			culture = {
				cultural_acceptance = {
					target = scope:target_character.culture
					value >= 25
				}
			}
		}
		multiply = 0.75
	}
	if = {
		limit = { knows_language_of_culture = scope:target_character.culture }
		multiply = define:NCharacterOpinion|CULTURE_ACCEPTANCE_MULT_KNOWS_LANGUAGE
	}
}

				add = cowed_marriage_not_secondary_reason_value
			}
		}
	}


	if = {
		limit = {
			NOT = { scope:recipient = scope:secondary_recipient }
			scope:recipient = {
				OR = {
					is_close_or_extended_family_of = scope:secondary_recipient
					has_relation_lover = scope:secondary_recipient
				}
				opinion = {
					target = scope:secondary_recipient
					value > 10
				}
			}
		}
		add = {
			value = scope:opinion_of_secondary_recipient
			multiply = {
				if = {
					limit = {
						scope:actor = {
							has_government = landless_adventurer_government
						}
					}
					value = -1
				}
				else = {
					value = -0.15
				}
			}
			if = {
				limit = { scope:opinion_of_secondary_recipient <= 0 }
				floor = yes
			}
			else = {
				ceiling = yes
			}
		}
	}


	if = {
		limit = {
			scope:recipient.faith = {
				NOT = {
					faith_allows_marriage_consanguinity_trigger = { CHARACTER_1 = scope:secondary_recipient CHARACTER_2 = scope:secondary_actor }
				}
			}
		}
		add = -1000
	}


	if = {
		limit = {
			scope:secondary_recipient = {
				effective_age > 20
				OR = {
					dynasty = scope:recipient.dynasty
					is_close_or_extended_family_of = scope:recipient
				}
			}
		}
		add = {
			value = scope:secondary_recipient.effective_age
			subtract = 20
		}
	}


	if = {
		limit = {
			exists = scope:secondary_recipient.inspiration

		}
		add = -100
	}


	scope:actor = {
		if = {
			limit = {
				has_royal_court = yes
				has_dlc_feature = royal_court
			}
			if = {
				limit = { court_grandeur_current_level > court_grandeur_minimum_expected_level }
				add = ep1_court_grandeur_marriage_acceptance_value
			}
			else_if = {
				limit = { court_grandeur_current_level < court_grandeur_minimum_expected_level }
				add = ep1_court_grandeur_marriage_acceptance_value_penalty
			}
		}
	}


	if = {
		limit = {
			scope:recipient.culture = {
				has_cultural_parameter = dislikes_marrying_outside_of_culture
				this != scope:actor.culture
			}
		}
		add = -50
	}


	if = {
		limit = {
			scope:actor.house ?= {
				has_variable = intermarrying_with_var
				var:intermarrying_with_var = scope:recipient.house
			}
		}
		add = 20
	}


	if = {
		limit = {
			scope:actor = {
				any_character_struggle = {
					involvement = involved
					is_struggle_parameter_active_secondary_character_involvement_involved_trigger = {
						PARAMETER = involved_form_alliances_with_each_other_easier
						CHAR = scope:recipient
					}
				}
			}
		}
		add = 25
	}
	if = {
		limit = {
			OR = {
				AND = {

					scope:actor = {
						is_ai = yes
						any_character_struggle = {
							involvement = involved
							is_secondary_character_involvement_neither_trigger = { CHAR = scope:recipient }
						}
					}
					scope:recipient = {
						is_ai = yes
					}
				}
				AND = {

					scope:recipient = {
						is_ai = yes
						any_character_struggle = {
							involvement = involved
							is_secondary_character_involvement_neither_trigger = { CHAR = scope:actor }
						}
					}
					scope:actor = {
						is_ai = yes
					}
				}
			}
		}
		multiply = {
			value = 0
			if = {
				limit = {
					OR = {
						AND = {
							scope:actor = {
								any_character_struggle = {
									involvement = involved
									phase_has_catalyst = catalyst_forming_alliance_with_major_uninvolved_power
								}
							}
							scope:recipient = {
								realm_size <= medium_realm_size
							}
						}
						AND = {
							scope:recipient = {
								any_character_struggle = {
									involvement = involved
									phase_has_catalyst = catalyst_forming_alliance_with_major_uninvolved_power
								}
							}
							scope:actor = {
								realm_size <= medium_realm_size
							}
						}
					}
				}
				add = 0.2
			}
			if = {
				limit = {
					OR = {
						AND = {
							scope:actor = {
								has_trait = fp3_struggle_detractor
								any_character_struggle = {
									involvement = involved
									phase_has_catalyst = catalyst_forming_alliance_between_detractor_uninvolved_rulers
								}
							}
							scope:recipient = {
								always = no
							}
						}
						AND = {
							scope:recipient = {
								has_trait = fp3_struggle_detractor
								any_character_struggle = {
									involvement = involved
									phase_has_catalyst = catalyst_forming_alliance_between_detractor_uninvolved_rulers
								}
							}
							scope:actor = {
								always = no
							}
						}
					}
				}
				add = 0.2
			}
		}
	}
	if = {
		limit = {
			OR = {
				AND = {
					scope:actor = {
						is_ai = yes
						any_character_struggle = {
							involvement = involved
							is_secondary_character_involvement_interloper_trigger = { CHAR = scope:recipient }
						}
					}
					scope:recipient = {
						is_ai = yes
					}
				}
				AND = {
					scope:recipient = {
						is_ai = yes
						any_character_struggle = {
							involvement = involved
							is_secondary_character_involvement_interloper_trigger = { CHAR = scope:actor }
						}
					}
					scope:actor = {
						is_ai = yes
					}
				}
			}
		}
		multiply = 0.5
	}


	if = {
		limit = {
			scope:recipient = {
				is_tributary_of_suzerain_or_above = scope:actor
			}
		}
		add = 50
	}
	if = {
		limit = {
			scope:recipient = {
				is_obedient_to = scope:actor
			}
		}
		add = 100
	}


	if = {
		limit = {
			scope:actor = {
				government_has_flag = government_is_nomadic
			}
			scope:recipient = {
				is_ruler = yes
				NOT = { government_has_flag = government_is_nomadic }
				NAND = {
					is_vassal_or_below_of = scope:actor
					is_obedient_to = scope:actor
				}
			}
		}
		if = {
			limit = {
				scope:recipient = {
					OR = {
						government_has_flag = government_is_tribal
						is_vassal_or_below_of = scope:actor
						is_obedient_to = scope:actor
					}
				}
			}
			add = -50
		}
		else = {
			add = -100
		}
	}


	if = {
		limit = {
			scope:recipient = {
				government_has_flag = government_is_nomadic
			}
			scope:actor = {
				prowess < 14
			}
		}
		add = {
			value = 14
			subtract = scope:actor.prowess
			multiply = -4
		}
	}


	if = {
		limit = {
			scope:recipient = {
				government_has_flag = government_is_nomadic
			}
		}
		if = {
			limit = {
				scope:actor = {
					has_realm_law = nomadic_authority_1
				}
			}
			add = -40
		}
		else_if = {
			limit = {
				scope:actor = {
					has_realm_law = nomadic_authority_2
				}
			}
			add = -20
		}
		else_if = {
			limit = {
				scope:actor = {
					has_realm_law = nomadic_authority_4
				}
			}
			add = 10
		}
		else_if = {
			limit = {
				scope:actor = {
					has_realm_law = nomadic_authority_5
				}
			}
			add = 20
		}
	}


	if = {
		limit = {
			scope:recipient = {
				is_ai = no
			}
			scope:secondary_recipient = {
				NOR = {
					is_close_or_extended_family_of = scope:recipient
					accou_is_same_dynasty_trigger = { CHARACTER = scope:recipient }
				}
			}
		}
		add = -5000
	}


	if = {
		limit = {
			scope:secondary_recipient.house ?= {
				NOT = { scope:secondary_actor.house ?= this }
				any_house_relation = {
					has_house_relation_parameter = increased_marriage_acceptance
					any_relation_house = { scope:secondary_actor.house ?= this }
				}
			}
		}
		add = house_relation_increased_marriage_acceptance
	}
	if = {
		limit = {
			scope:secondary_recipient.house ?= {
				NOT = { scope:secondary_actor.house ?= this }
				any_house_relation = {
					has_house_relation_parameter = decreased_marriage_acceptance
					any_relation_house = { scope:secondary_actor.house ?= this }
				}
			}
		}
		add = house_relation_decreased_marriage_acceptance
	}


	if = {
		limit = {
			scope:secondary_recipient.culture = { has_cultural_parameter = reduced_marriage_acceptance_outside_culture }
			NOT = { scope:secondary_actor.culture = scope:secondary_recipient.culture }

			trigger_if = {
				limit = {
					scope:recipient.primary_title = title:h_china
					scope:actor.overlord = scope:recipient
				}
				NOT = { scope:actor = { subject_contract_has_flag = celestial_investiture_marriage } }
			}
		}
		add = {
			value = -25
			if = {
				limit = {
					NOT = { scope:secondary_actor.culture = { has_same_culture_heritage = scope:secondary_recipient.culture } }
				}
				add = -25
			}
		}
	}


	if = {
		limit = {
			trigger_if = {
				limit = { exists = scope:actor }
				scope:actor = { is_ai = yes }
			}
			trigger_else = {
				scope:secondary_actor = { is_ai = yes }
			}
			scope:secondary_actor.culture = { has_cultural_parameter = reduced_marriage_acceptance_outside_culture }
			NOT = { scope:secondary_actor.culture = scope:secondary_recipient.culture }
		}
		add = {
			value = -100
			if = {
				limit = {
					NOT = { scope:secondary_actor.culture = { has_same_culture_heritage = scope:secondary_recipient.culture } }
				}
				add = -100
			}
		}
	}


	if = {
		limit = {
			tgp_distant_realm_marriage_trigger = {
				RECIPIENT_SPOUSE = scope:secondary_recipient
				ACTOR_SPOUSE = scope:secondary_actor
			}

			trigger_if = {
				limit = {
					scope:recipient.primary_title = title:h_china
					scope:actor.overlord = scope:recipient
				}
				NOT = { scope:actor = { subject_contract_has_flag = celestial_investiture_marriage } }
			}
		}
		add = {
			value = -1
			multiply = {
				value = "scope:from.squared_distance(scope:to)"
				divide = 20000
			}
			min = -50
			max = -1
		}
	}


	if = {
		limit = {
			tgp_distant_realm_marriage_trigger = {
				RECIPIENT_SPOUSE = scope:secondary_actor
				ACTOR_SPOUSE = scope:secondary_recipient
			}
		}
		add = {
			value = -1
			multiply = {
				value = "scope:from.squared_distance(scope:to)"
				divide = 40000
			}
			min = -50
			max = -1
		}
	}

	if = {
		limit = {
			scope:recipient.primary_title = title:h_china
			scope:actor = {
				overlord = scope:recipient
				subject_contract_has_flag = celestial_investiture_marriage
			}
		}
		add = {
			add = scope:actor.subject_standing
			divide = 4
		}
	}


	if = {
		limit = {
			trigger_if = {
				limit = { exists = scope:actor }
				scope:actor = { is_ai = yes }
			}
			trigger_else = {
				scope:secondary_actor = { is_ai = yes }
			}
			tgp_distant_realm_marriage_trigger = {
				RECIPIENT_SPOUSE = scope:secondary_actor
				ACTOR_SPOUSE = scope:secondary_recipient
			}
		}
		add = {
			value = -1
			multiply = {
				value = "scope:from.squared_distance(scope:to)"
				divide = 20000
			}
			multiply = 2
			min = -100
			max = -1
		}
	}

	if = {
		limit = {
			scope:secondary_recipient = { tgp_is_in_ceremonial_house_trigger = yes }
			scope:secondary_actor.house.var:petition_liege_wedding_permission_flag ?= scope:secondary_recipient.house
		}
		add = 50
	}

	if = {
		limit = {
		scope:secondary_recipient = { tgp_is_in_ceremonial_house_trigger = yes }
			NOR = {
				tgp_shared_ceremonial_house_trigger = {
					COMPARE_1 = scope:secondary_recipient
					COMPARE_2 = scope:secondary_actor
				}
				scope:secondary_actor.house.var:petition_liege_wedding_permission_flag ?= scope:secondary_recipient.house
			}
			scope:secondary_recipient.top_liege.primary_title.var:administrative_ui_special_title ?= {
				tier > scope:secondary_actor.highest_held_title_tier
			}
		}
		add = {
			value = -25
			multiply = {
				value = scope:secondary_recipient.top_liege.primary_title.var:administrative_ui_special_title.holder.dynasty.dynasty_prestige_level
				subtract = scope:secondary_actor.dynasty.dynasty_prestige_level
			}
			max = -25
		}
	}
	if = {
		limit = {
			scope:secondary_recipient = { tgp_is_in_ceremonial_house_trigger = yes }
			NOR = {
				tgp_shared_ceremonial_house_trigger = {
					COMPARE_1 = scope:secondary_recipient
					COMPARE_2 = scope:secondary_actor
				}
				scope:secondary_actor.house.var:petition_liege_wedding_permission_flag ?= scope:secondary_recipient.house
				scope:secondary_actor = {
					any_held_title = { is_noble_family_title = yes }
				}
			}
		}
		add  = -50
	}
	if = {
		limit = {
			NOT = {
				tgp_shared_ceremonial_house_trigger = {
					COMPARE_1 = scope:secondary_recipient
					COMPARE_2 = scope:secondary_actor
				}
			}
			scope:secondary_actor = { is_married = yes }
			scope:secondary_recipient = { tgp_is_in_ceremonial_house_trigger = yes }
		}
		add = -50
	}


	if = {
		limit = {
			OR = {
				has_game_rule = hard_difficulty
				has_game_rule = very_hard_difficulty
			}
			scope:actor = {
				is_ai = no
			}
			scope:secondary_actor = {
				NOR = {
					is_close_family_of = scope:actor
					this = scope:actor
				}
			}
			scope:secondary_recipient = {
				is_ruler = no
				OR = {
					diplomacy > 12
					stewardship > 12
					martial > 12
					learning > 12
					intrigue > 12
					prowess > 12
				}
			}
		}
		add = -2000
	}


}

	if = {
		limit = { has_game_rule = accou_marriage_candidate_quality_threshold_very_high }
		value = 10
	}
	else_if = {
		limit = { has_game_rule = accou_marriage_candidate_quality_threshold_high }
		value = 0
	}
	else_if = {
		limit = { has_game_rule = accou_marriage_candidate_quality_threshold_medium }
		value = -10
	}
	else_if = {
		limit = { has_game_rule = accou_marriage_candidate_quality_threshold_low }
		value = -20
	}
	else_if = {
		limit = { has_game_rule = accou_marriage_candidate_quality_threshold_very_low }
		value = -30
	}
	else = {
		value = 0
	}
}
