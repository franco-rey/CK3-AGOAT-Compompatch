					save_scope_as = zzz_great_councils_pregnant
				}
			}
		}

			limit = { scope:transfer_type = flag:inheritance }

				list = zzz_great_councils_electors
				
				trigger_event = zzz_great_councils_event.0001
			}
		}
		else = {
			scope:zzz_great_councils_emperor = {
				remove_from_list = zzz_great_councils_candidates
				remove_variable = zzz_great_councils_votes
			}
			root = { add_character_flag = zzz_great_councils_depose }
			every_in_list = {
				list = zzz_great_councils_electors
				
				trigger_event = zzz_great_councils_event.0001
			}
		}
	}
