on_combat_end_winner = {
    on_actions = {
        on_add_vet_modifier
    }
}

on_combat_end_loser = {
    on_actions = {
        on_add_vet_modifier
    }
}

on_add_vet_modifier = {
    effect = {
        every_side_knight = {
            if = {
                limit = {
                    is_ruler = no
                    # AGOAT: upstream's `NOT = { father = { is_army_owner = root.army_owner } }` uses a trigger that does not exist, which made the
                    # engine reject this on_action ("Unknown trigger: is_army_owner"), so no knight ever received the veteran modifier. Dropped.
                }
                add_character_modifier = veteran_modifier
            }
        }
    }
}
