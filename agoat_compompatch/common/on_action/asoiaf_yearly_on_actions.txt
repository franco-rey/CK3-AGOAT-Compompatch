asoiaf_mance_rayder_desertion = {
	trigger = {
		asoiaf_historical_trigger = yes
		root = {
			exists = character:Rayder_1
			character:Rayder_1 = { is_alive = yes }
			this = character:Rayder_1
			OR = {
				has_trait = nightswatch
				has_trait = nightswatch_historical
			}
			NOT = {
				is_imprisoned = yes
				is_incapable = yes
			}
			is_ai = yes
			NOT = { has_trait = conqueror }
			is_lowborn = yes
		}
		current_date >= 8289.1.1


		title:e_beyond_the_wall = { is_title_created = no }
	}
	effect = {
		root = {
			set_house = house:house_Rayder
			add_trait = conqueror
			if = {
				limit = {
					has_game_rule = extreme_conquerors_bonuses
					NOT = { has_character_modifier = ai_extreme_conqueror_modifier }
				}
				add_character_modifier = { modifier = ai_extreme_conqueror_modifier }
			}
			set_character_faith = faith:old_gods_btw_fnf
			set_culture = culture:forestman
			if = {
				limit = { has_trait = nightswatch }
				remove_trait = nightswatch
			}
			if = {
				limit = { has_trait = nightswatch_historical }
				remove_trait = nightswatch_historical
			}
			if = {
				limit = { has_trait = lifestyle_nw_builder }
				remove_trait = lifestyle_nw_builder
			}
			if = {
				limit = { has_trait = lifestyle_nw_ranger }
				remove_trait = lifestyle_nw_ranger
			}
			if = {
				limit = { has_trait = lifestyle_nw_steward }
				remove_trait = lifestyle_nw_steward
			}
			add_trait = deserter
			if = {
				limit = {
					OR = {
						is_landed = yes
						is_landless_ruler = yes
					}
				}
				agot_depose_effect = yes
			}
			title:k_first_forest ?= {
				random_de_jure_county = {
					limit = {
						holder = { is_ai = yes }
					}
					save_scope_as = asoiaf_mance_new_county
				}
			}
			if = {
				limit = {
					scope:asoiaf_mance_new_county.holder = { government_has_flag = government_is_uninteractable }
				}
				ai_colonization_effect = { WILDERNESS = scope:asoiaf_mance_new_county ROOT_SCOPE = root }
			}
			else = {
				create_title_and_vassal_change = {
					type = conquest
					save_scope_as = change
					add_claim_on_loss = yes
				}
				scope:asoiaf_mance_new_county = {
					change_title_holder = {
						holder = root
						change = scope:change
					}
				}
				resolve_title_and_vassal_change = scope:change
			}
			create_title_and_vassal_change = {
				type = created
				save_scope_as = change
				add_claim_on_loss = yes
			}
			title:e_beyond_the_wall = {
				change_title_holder = {
					holder = root
					change = scope:change
				}
			}
			resolve_title_and_vassal_change = scope:change
			change_government = tribal_government
			add_prestige = 3000
			add_gold = 100
			add_legitimacy = 100
			spawn_army = {
				name = asoiaf_wildling_army
				men_at_arms = {
					type = giant_regiment
					stacks = 20
				}
				men_at_arms = {
					type = light_footmen
					stacks = 30
				}
				men_at_arms = {
					type = bowmen
					stacks = 30
				}
				men_at_arms = {
					type = battering_ram
					stacks = 1
				}
				levies = 4000
				location = root.location
				inheritable = yes
			}
			create_character_memory = { type = asoiaf_mance_rayder_desertion_memory }
			if = {
				limit = {
					exists = title:k_the_wall.holder
				}
				set_relation_rival = { reason = asoiaf_mance_lord_commander_rival_desc target = title:k_the_wall.holder }
			}
		}
		every_living_character = {
			limit = {
				OR = {
					is_landed = yes
					is_landless_ruler = yes
				}
				this.location = { geographical_region = world_westeros }
				NOT = { this = character:Rayder_1 }
			}
			trigger_event = asoiaf_maintenance_events.0009
		}
	}
}
