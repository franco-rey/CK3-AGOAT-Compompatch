	first_valid_on_action = {
		standard_abduct_method_setup
	}
}



standard_abduct_method_setup = {
	random_events = {
		100 = abduct_outcome.1001
		100 = abduct_outcome.1002
		100 = abduct_outcome.1003
	}
}

abduct_succeeded = {
	trigger = { exists = scope:scheme }
	first_valid = {
		abduct_outcome.2001
		abduct_outcome.2002
		abduct_outcome.2003
	}
	effect = {
		if = {
			limit = { exists = scope:owner }
			scope:owner = {
				random = {
					chance = 2
					modifier = {
						factor = accolade_progress
					}
					modifier = {
						NOT = {
							any_knight = {
								can_unlock_accolade_attribute_trigger = {
									ATTRIBUTE = scoundrel
								}
							}
						}
						factor = 0
					}
					save_scope_as = accolade_liege
					set_variable = {
						name = accolade_progress
						value = 0
					}
					random_knight = {
						limit = {
							can_unlock_accolade_attribute_trigger = {
								ATTRIBUTE = scoundrel
							}
						}
						weight = {
							base = 1
							modifier = {
								add = 50
								is_acclaimed = yes
							}
						}
						save_scope_as = accolade_knight
					}
					send_interface_message = {
						type = msg_accolade_eligibility
						title = accolade_scoundrel_unlock.t
						left_icon = scope:accolade_knight
						right_icon = scope:accolade_liege
						custom_tooltip = accolade_scoundrel_unlock.tt
						scope:accolade_knight = {
							set_variable = {
								name = scoundrel_attribute_unlock
								value = yes
							}
						}	
					}
				}
			}
		}
	}
}
