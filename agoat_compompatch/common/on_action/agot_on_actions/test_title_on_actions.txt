asoiaf_on_title_inheritance_rhaenyra_kills_siblings = {
	trigger = {
		scope:title = { this = title:h_the_iron_throne }
		scope:previous_holder = { has_inactive_trait = asoiaf_Targaryen_60_trait }
		root = { has_inactive_trait = asoiaf_Targaryen_63_trait }
		root = {
			father = {
				any_child = {
					AND = {
						is_male = yes
						is_alive = yes
						NOR = {
							has_trait = maester
							has_trait = archmaester
							has_trait = kingsguard
							has_trait = most_devout_member
							has_trait = septon
							has_trait = devoted
							has_trait = red_priest
							has_trait = drowned_man
							has_trait = nightswatch
							has_trait = bastard
						}
						NOR = {
							is_spouse_of = root
							AND = {
								is_spouse_of_even_if_dead = root
								any_child = {
									AND = {
										mother = root
										is_alive = yes
										NOR = {
											has_trait = maester
											has_trait = archmaester
											has_trait = kingsguard
											has_trait = most_devout_member
											has_trait = septon
											has_trait = devoted
											has_trait = red_priest
											has_trait = drowned_man
											has_trait = nightswatch
											has_trait = bastard
											has_trait = disinherited
										}
									}
								}
							}
						}
					}
				}
			}
		}
	}
	effect = {
		root = {
			trigger_event = { id = asoiaf_maintenance_events.0006 days = 2 }
		}
	}
}

asoiaf_iron_throne_lord_paramount_contract_fix = {
	trigger = {
		has_game_rule = asoiaf_lord_paramount_contract_fix_on

			OR = {
				this = title:e_the_north
				this = title:e_the_riverlands

		}

asoiaf_on_title_inheritance_lannister_baratheon_coa = {
	trigger = {
		asoiaf_historical_trigger = yes
		scope:title = {
			OR = {
				this = title:h_the_iron_throne
				this = title:e_the_westerlands
				this = title:e_the_crownlands
			}
		}
		root = {
			dynasty = dynasty:dynn_Baratheon
			father = { has_inactive_trait = asoiaf_Baratheon_2_trait }
			mother = { has_inactive_trait = asoiaf_Lannister_6_trait }
			NOT = { real_father = { has_inactive_trait = asoiaf_Baratheon_2_trait } }
		}
		NOR = {
			is_target_in_global_variable_list = {
				name = asoiaf_baratheon_kl_unavailable
				target = flag:asoiaf_baratheon_cadet_kl
			}
			house ?= house:house_BaratheonKL
		}
	}
	effect = {
		if = {
			limit = { has_game_rule = asoiaf_new_clothes_westerlands_rule_on }
			trigger_event = { id = asoiaf_maintenance_events.0014 days = 30 }
		}
		set_house = house:house_BaratheonKL
		set_culture = culture:crownlander
		add_trait = disputed_heritage
		if = {
			limit = {
				age < 16
				root.mother = { is_courtier_of = root }
			}
			designate_diarch = root.mother
			set_diarch = root.mother
			set_diarchy_type = regency
		}
		every_child = {
			if = {
				limit = {
					dynasty = dynasty:dynn_Baratheon
				}
				set_culture = culture:crownlander
				set_house = house:house_BaratheonKL
				trigger_event = { id = agot_coa_events.1 }
				every_child = {
					if = {
						limit = {
							dynasty = dynasty:dynn_Baratheon
						}
						set_culture = culture:crownlander
						set_house = house:house_BaratheonKL
						trigger_event = { id = agot_coa_events.1 }
					}
				}
			}
		}
		every_close_family_member = {
			if = {
				limit = {
					dynasty = dynasty:dynn_Baratheon
					father = { has_inactive_trait = asoiaf_Baratheon_2_trait }
					mother = { has_inactive_trait = asoiaf_Lannister_6_trait }
					NOT = {
						has_trait = bastard
					}
				}
				set_culture = culture:crownlander
				set_house = house:house_BaratheonKL
				add_trait = disputed_heritage
				trigger_event = { id = agot_coa_events.1 }
				every_child = {
					if = {
						limit = {
							dynasty = dynasty:dynn_Baratheon
						}
						set_culture = culture:crownlander
						set_house = house:house_BaratheonKL
						trigger_event = { id = agot_coa_events.1 }
						every_child = {
							if = {
								limit = {
									dynasty = dynasty:dynn_Baratheon
								}
								set_culture = culture:crownlander
								set_house = house:house_BaratheonKL
								trigger_event = { id = agot_coa_events.1 }
							}
						}
					}
				}
			}
		}
		add_to_global_variable_list = {
			name = asoiaf_baratheon_kl_unavailable
			target = flag:asoiaf_baratheon_cadet_kl
		}
		trigger_event = { id = agot_coa_events.1 }
		dynasty:dynn_Lannister = {
			every_dynasty_member = {
				limit = {
					has_inactive_trait = asoiaf_Lannister_1_trait
					is_alive = yes
				}
				save_scope_as = asoiaf_Lannister_1
			}
		}
		if = {
			limit = {
				exists = scope:asoiaf_Lannister_1
			}
			create_alliance = {
				target = scope:asoiaf_Lannister_1
				allied_through_owner = ROOT
				allied_through_target = scope:asoiaf_Lannister_1
			}
			scope:asoiaf_Lannister_1 = {
				add_opinion = {
					target = ROOT
					modifier = perk_negotiated_alliance_opinion
				}
			}
		}
	}
}

asoiaf_on_title_inheritance_frey_rosby_cadet = {
	trigger = {
		NOT = { exists = global_var:asoiaf_frey_rosby_cadet_branch_founded_variable }


		scope:title = {
			OR = {
				this = title:d_rosby
				this = title:c_rosby
			}
		}
		root = {
			dynasty = dynasty:dynn_Frey
			OR = {
				this = character:Frey_78
				this = character:Frey_79
				this = character:Frey_80
				this = character:Frey_81
				this = character:Frey_82
				AND = {
					father ?= character:Frey_78
					house ?= character:Frey_78.house
				}
				AND = {
					father ?= character:Frey_79
					house ?= character:Frey_79.house
				}
				AND = {
					father ?= character:Frey_80
					house ?= character:Frey_80.house
				}
				AND = {
					father ?= character:Frey_81
					house ?= character:Frey_81.house
				}
				AND = {
					mother ?= character:Frey_82
					house ?= character:Frey_82.house
				}
			}
		}
	}
	effect = {
		set_global_variable = asoiaf_frey_rosby_cadet_branch_founded_variable
		root = {
			house = { save_scope_as = asoiaf_founders_old_house }
			set_house = house:house_Frey_Rosby
			trigger_event = { id = agot_coa_events.1 }
			add_prestige = medium_prestige_gain
			dynasty = {
				every_dynasty_member = {
					limit = {
						any_ancestor = {
							even_if_dead = yes
							this = character:Rosby_3
						}
						house = scope:asoiaf_founders_old_house
					}
					set_house = house:house_Frey_Rosby
					trigger_event = { id = agot_coa_events.1 }
				}
			}
		}
		character:Frey_79 ?= {
			if = {
				limit = { has_character_flag = has_personal_coa }
				agot_prune_personal_coa = yes
			}
		}
	}
}

asoiaf_on_title_inheritance_harrold_hardyng_becomes_arryn = {
	trigger = {
		asoiaf_historical_trigger = yes
		scope:title = {
			OR = {
				this = title:e_the_vale
				this = title:e_the_vale
			}
		}
		root = character:Hardyng_5
		scope:previous_holder = {
			OR = {
				dynasty = dynasty:dynn_Arryn
				has_title = title:h_the_iron_throne
			}
		}
	}
	effect = {
		character:Hardyng_5 = {
			house ?= {
				character:Hardyng_5 = { set_house = character:Arryn_15.house }
				every_house_member = {
					limit = {
						any_ancestor = {
							this = character:Hardyng_5
						}
					}
					set_house = character:Arryn_15.house
				}
			}
		}
		character:Hardyng_5.mother = {
			set_house = character:Arryn_15.house
		}
	}
}

asoiaf_on_title_inheritance_aegon_I_crown_creation = {
	trigger = {
		AND = {
			this = { has_inactive_trait = asoiaf_Targaryen_27_trait }
			NOT = { this = character:Targaryen_27 }
			OR = {
				primary_title.tier = tier_hegemony
				AND = {
					primary_title.tier = tier_empire
					agot_is_independent_ruler = yes
				}
			}
			asoiaf_aegon_I_targaryen_crown_trigger = yes
			NOT = { has_government = nights_watch_government }
			NOT = { has_government = silent_sisterhood_government }
			NOT = { has_government = theocracy_government }
		}
	}
	effect = {
		this = { save_scope_as = asoiaf_Targaryen_27 }
		scope:asoiaf_Targaryen_27 ?= {
			agot_create_artifact_aegon_i_crown_effect = { OWNER = this CREATOR = this }
			trigger_event = { id = asoiaf_equip_crown_events.1 days = 1 }
		}
	}
}
