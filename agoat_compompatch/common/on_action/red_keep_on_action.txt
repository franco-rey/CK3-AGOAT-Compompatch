rk_on_gk_loyalty = {
	effect = {
		if = {
			limit = {
				is_ai = no
				has_title = title:c_kings_landing
				title:b_goldguard_heights.holder = {
					opinion = {
						target = root
						value >= 50
					}
				}	
				NOT = {
					has_title = title:b_goldguard_heights	
					has_variable = rk_loyal_commander					
				}	
			}
			trigger_event = rk_crime.0001
			set_variable = rk_loyal_commander
			remove_variable = rk_disloyal_commander
			remove_variable = rk_neutral_commander
		}
		else_if = { 
			limit = {
				is_ai = no
				has_title = title:c_kings_landing
				NOT = {
					title:b_goldguard_heights.holder = {
						opinion = {
							target = root
							value >= 0
						}
					}	
				}
				NOT = {
					has_title = title:b_goldguard_heights
					has_variable = rk_disloyal_commander					
				}					
			}
			trigger_event = rk_crime.0002	
			set_variable = rk_disloyal_commander
			remove_variable = rk_loyal_commander		
			remove_variable = rk_neutral_commander			
		}
		else_if = { 
			limit = {
				is_ai = no
				has_title = title:c_kings_landing
				title:b_goldguard_heights.holder = {
					opinion = {
						target = root
						value = { 1 49 }
					}	
				}
				NOT = {
					has_title = title:b_goldguard_heights
					has_variable = rk_neutral_commander
				}					
			}
			trigger_event = rk_crime.0003	
			remove_variable = rk_disloyal_commander
			remove_variable = rk_loyal_commander
			set_variable = rk_neutral_commander
		}	
		else_if = { 
			limit = {
				is_ai = no
				NOT = {
					has_title = title:c_kings_landing				
				}
				OR = {
					has_variable = rk_disloyal_commander
					has_variable = rk_neutral_commander
					has_variable = rk_loyal_commander					
				}					
			}
			trigger_event = rk_crime.0004	
		}			
	}
}

on_king_death_kingsguard = {
	trigger = {
		OR = {
			has_character_modifier = rk_kingsguard_honor_1
			has_character_modifier = rk_kingsguard_honor_2
			has_character_modifier = rk_kingsguard_honor_3
			has_character_modifier = rk_kingsguard_honor_4
			has_character_modifier = rk_kingsguard_honor_5		
		}
		exists = player_heir	
		has_title = title:c_kings_landing
	}

	effect = {
		if = {
			limit = {
				has_character_modifier = rk_kingsguard_honor_1
			}
			player_heir = {				
				add_character_modifier = rk_kingsguard_honor_1					
			}			
		}
		else_if = {
			limit = {
				has_character_modifier = rk_kingsguard_honor_2
			}
			player_heir = {				
				add_character_modifier = rk_kingsguard_honor_2					
			}			
		}
		else_if = {
			limit = {
				has_character_modifier = rk_kingsguard_honor_3
			}
			player_heir = {				
				add_character_modifier = rk_kingsguard_honor_3					
			}			
		}
		else_if = {
			limit = {
				has_character_modifier = rk_kingsguard_honor_4
			}
			player_heir = {				
				add_character_modifier = rk_kingsguard_honor_4					
			}			
		}
		else_if = {
			limit = {
				has_character_modifier = rk_kingsguard_honor_5
			}
			player_heir = {				
				add_character_modifier = rk_kingsguard_honor_5					
			}			
		}
		else = {
			player_heir = {				
				add_character_modifier = rk_kingsguard_honor_3				
			}		
		}
	}
}

on_king_death = {
	trigger = {
		OR = {
			has_character_modifier = rk_seignorage_profits	
			has_character_modifier = rk_pure_coinage
			has_character_modifier = rk_strong_coinage
			has_character_modifier = rk_common_coinage
			has_character_modifier = rk_debased_coinage
			has_character_modifier = rk_worthless_coinage		
		}
		exists = player_heir	
		has_title = title:c_kings_landing
	}

	effect = {
			player_heir = {				
				add_character_modifier = rk_seignorage_profits					
			}	
		if = {
			limit = {
				has_character_modifier = rk_pure_coinage
			}
			player_heir = {				
				add_character_modifier = rk_pure_coinage					
			}			
		}
		else_if = {
			limit = {
				has_character_modifier = rk_strong_coinage
			}
			player_heir = {				
				add_character_modifier = rk_strong_coinage					
			}			
		}
		else_if = {
			limit = {
				has_character_modifier = rk_common_coinage
			}
			player_heir = {				
				add_character_modifier = rk_common_coinage					
			}			
		}
		else_if = {
			limit = {
				has_character_modifier = rk_debased_coinage
			}
			player_heir = {				
				add_character_modifier = rk_debased_coinage					
			}			
		}
		else_if = {
			limit = {
				has_character_modifier = rk_worthless_coinage
			}
			player_heir = {				
				add_character_modifier = rk_worthless_coinage					
			}			
		}		
	}
}
