interactive_bannermen_mode_decision = {
	desc = interactive_bannermen_mode_decision_desc
	selection_tooltip = interactive_bannermen_mode_decision_tooltip
	confirm_text = interactive_bannermen_mode_decision_confirm
	decision_group_type = interactive_vassals
	is_shown = {
		has_game_rule = bannermen_mode_enabled
		highest_held_title_tier >= tier_duchy
		is_landed = yes
		is_ruler = yes
		any_vassal = {
			interactive_bannermen_scope_trigger = {
				LIEGE = root
			}
		}
		OR = {
			is_ai = no
			is_at_war = no
			any_character_war = {
				interactive_bannermen_callable_war_trigger = {
					LIEGE = root
				}
			}
		}
	}
	is_valid = {
		is_at_war = yes
	}
	cooldown = {
		months = 12
	}
	ai_check_interval = 2
	ai_potential = {
		has_game_rule = bannermen_mode_enabled
		is_at_war = yes
		highest_held_title_tier >= tier_duchy
		is_landed = yes
		is_ruler = yes
		any_character_war = {
			interactive_bannermen_callable_war_trigger = {
				LIEGE = root
			}
			interactive_bannermen_worth_calling_trigger = {
				LIEGE = root
			}
			OR = {
				OR = {
					primary_attacker = root
					primary_defender = root
				}
				AND = {
					has_game_rule = bannermen_extended_mode_1_enabled
					root = {
						highest_held_title_tier >= tier_empire
					}
					NOT = {
						primary_attacker = root
					}
					NOT = {
						primary_defender = root
					}
				}
				AND = {
					has_game_rule = bannermen_extended_mode_2_enabled
					root = {
						highest_held_title_tier >= tier_kingdom
					}
					NOT = {
						primary_attacker = root
					}
					NOT = {
						primary_defender = root
					}
				}
				AND = {
					has_game_rule = bannermen_extended_mode_3_enabled
					root = {
						highest_held_title_tier >= tier_duchy
					}
					NOT = {
						primary_attacker = root
					}
					NOT = {
						primary_defender = root
					}
				}
				AND = {
					interactive_agot_megawar_empire_override_trigger = {
						LIEGE = root
					}
					NOT = {
						primary_attacker = root
					}
					NOT = {
						primary_defender = root
					}
				}
			}
		}
	}
	ai_will_do = {
		base = 1000
	}
	effect = {
		save_temporary_scope_as = liege
		scope:liege = {
			interactive_bannermen_preview_effect = yes
		}
		if = {
			limit = {
				scope:liege = {
					is_ai = no
				}
			}
			scope:liege = {
				custom_description_no_bullet = {
					text = interactive_bannermen_tooltip_intro_text
				}
				if = {
					limit = {
						OR = {
							has_game_rule = vassal_distance_land
							has_game_rule = vassal_distance_land_or_sea
						}
					}
					custom_description_no_bullet = {
						text = interactive_bannermen_distance_note_text
					}
				}
				custom_description_no_bullet = {
					text = interactive_bannermen_summary_answer_text
					subject = scope:liege
					value = miv_wardec_answering_count
				}
				custom_description_no_bullet = {
					text = interactive_bannermen_summary_answer_strength_text
					subject = scope:liege
					value = miv_wardec_answering_strength
				}
				custom_description_no_bullet = {
					text = interactive_bannermen_summary_refuse_text
					subject = scope:liege
					value = miv_wardec_withheld_count
				}
				custom_description_no_bullet = {
					text = interactive_bannermen_wardec_refuse_strength_text
					subject = scope:liege
					value = miv_wardec_withheld_strength
				}
				custom_description_no_bullet = {
					text = interactive_bannermen_summons_betray_text
					subject = scope:liege
					value = miv_wardec_betraying_count
				}
				custom_description_no_bullet = {
					text = interactive_bannermen_summons_betray_strength_text
					subject = scope:liege
					value = miv_wardec_betraying_strength
				}
			}
		}
		hidden_effect = {
			save_temporary_scope_as = liege
			if = {
				limit = {
					scope:liege = {
						is_ai = no
					}
				}
				play_music_cue = "mx_cue_combat_2"
			}
			if = {
				limit = {
					OR = {
						AND = {
							scope:liege = {
								is_ai = no
							}
							has_game_rule = civil_war_indirect_vassals_enabled
							scope:liege = {
								any_character_war = {
									primary_defender = {
										is_liege_or_above_of = scope:liege
									}
									primary_attacker = scope:liege
								}
							}
						}
						scope:liege = {
							any_character_war = {
								interactive_bannermen_callable_war_trigger = {
									LIEGE = scope:liege
								}
							}
						}
					}
				}
				scope:liege = {
					save_scope_as = miv_bannermen_liege
					if = {
						limit = {
							any_character_war = {
								is_civil_war = no
								interactive_bannermen_callable_war_trigger = {
									LIEGE = scope:liege
								}
							}
						}
						random_character_war = {
							limit = {
								is_civil_war = no
								interactive_bannermen_callable_war_trigger = {
									LIEGE = scope:liege
								}
							}
							save_scope_as = miv_bannermen_war
						}
						every_in_list = {
							list = interactive_bannermen_unavailable
							save_temporary_scope_as = miv_toast_vassal
							if = {
								limit = {
									miv_bannermen_distance_trigger = {
										LIEGE = scope:liege
										VASSAL = scope:miv_toast_vassal
									}
								}
								trigger_event = {
									id = interactive.0011
									days = {
										60
										90
									}
								}
							}
						}
						every_in_list = {
							list = interactive_bannermen_refuse
							save_temporary_scope_as = miv_toast_vassal
							if = {
								limit = {
									miv_bannermen_distance_trigger = {
										LIEGE = scope:liege
										VASSAL = scope:miv_toast_vassal
									}
								}
								trigger_event = {
									id = interactive.0020
									days = {
										150
										180
									}
								}
							}
						}
					}
					every_in_list = {
						list = interactive_bannermen_serving
						add_to_list = interactive_accept
					}
					every_in_list = {
						list = interactive_bannermen_prompt
						add_to_list = interactive_accept
					}
					every_in_list = {
						list = interactive_bannermen_willing
						add_to_list = interactive_accept
					}
					every_in_list = {
						list = interactive_bannermen_reluctant
						add_to_list = interactive_accept
					}
					every_in_list = {
						list = interactive_bannermen_grudging
						add_to_list = interactive_accept
					}
					every_in_list = {
						list = interactive_bannermen_refuse
						add_to_list = interactive_refuse
					}
					every_in_list = {
						list = interactive_bannermen_betray
						add_to_list = interactive_refuse
					}
					every_character_war = {
						save_temporary_scope_as = war
						if = {
							limit = {
								scope:war = {
									any_war_attacker = {
										this = scope:liege
									}
								}
								scope:war = {
									interactive_bannermen_callable_war_trigger = {
										LIEGE = scope:liege
									}
								}
								OR = {
									scope:war = {
										primary_attacker = scope:liege
									}
									scope:liege = {
										is_ai = no
									}
									AND = {
										has_game_rule = bannermen_extended_mode_1_enabled
										scope:liege = {
											highest_held_title_tier >= tier_empire
										}
									}
									AND = {
										has_game_rule = bannermen_extended_mode_2_enabled
										scope:liege = {
											highest_held_title_tier >= tier_kingdom
										}
									}
									AND = {
										has_game_rule = bannermen_extended_mode_3_enabled
										scope:liege = {
											highest_held_title_tier >= tier_duchy
										}
									}
									scope:war = {
										interactive_agot_megawar_empire_override_trigger = {
											LIEGE = scope:liege
										}
									}
								}
							}
							every_in_list = {
								list = interactive_accept
								save_temporary_scope_as = vassal
								if = {
									limit = {
										OR = {
											scope:war = {
												interactive_agot_megawar_trigger = yes
											}
											miv_bannermen_distance_war_trigger = {
												LIEGE = scope:liege
												VASSAL = scope:vassal
												ENEMY = scope:war.primary_defender
											}
										}
										scope:vassal = {
											NOT = {
												is_at_war_with = scope:war.primary_attacker
											}
										}
										scope:vassal = {
											NOT = {
												is_at_war_with = scope:war.primary_defender
											}
										}
									}
									scope:vassal = {
										interactive_bannermen_summon_effect = yes
									}
								}
							}
						}
						else_if = {
							limit = {
								scope:war = {
									any_war_defender = {
										this = scope:liege
									}
								}
								scope:war = {
									interactive_bannermen_callable_war_trigger = {
										LIEGE = scope:liege
									}
								}
								OR = {
									scope:war = {
										primary_defender = scope:liege
									}
									scope:liege = {
										is_ai = no
									}
									AND = {
										has_game_rule = bannermen_extended_mode_1_enabled
										scope:liege = {
											highest_held_title_tier >= tier_empire
										}
									}
									AND = {
										has_game_rule = bannermen_extended_mode_2_enabled
										scope:liege = {
											highest_held_title_tier >= tier_kingdom
										}
									}
									AND = {
										has_game_rule = bannermen_extended_mode_3_enabled
										scope:liege = {
											highest_held_title_tier >= tier_duchy
										}
									}
									scope:war = {
										interactive_agot_megawar_empire_override_trigger = {
											LIEGE = scope:liege
										}
									}
								}
							}
							every_in_list = {
								list = interactive_accept
								save_temporary_scope_as = vassal
								if = {
									limit = {
										OR = {
											scope:war = {
												interactive_agot_megawar_trigger = yes
											}
											miv_bannermen_distance_war_trigger = {
												LIEGE = scope:liege
												VASSAL = scope:vassal
												ENEMY = scope:war.primary_attacker
											}
										}
										scope:vassal = {
											NOT = {
												is_at_war_with = scope:war.primary_attacker
											}
										}
										scope:vassal = {
											NOT = {
												is_at_war_with = scope:war.primary_defender
											}
										}
									}
									scope:vassal = {
										interactive_bannermen_summon_effect = yes
									}
								}
							}
						}
					}
					every_in_list = {
						list = interactive_refuse
						save_temporary_scope_as = vassal
						if = {
							limit = {
								miv_bannermen_distance_trigger = {
									LIEGE = scope:liege
									VASSAL = scope:vassal
								}
							}
							scope:vassal = {
								interactive_bannermen_refuse_effect = yes
							}
						}
					}
				}
			}
		}
	}
	widget = {
		gui = "decision_view_widget_miv_muster_roll"
		controller = decision_option_list_controller
		show_from_start = yes
		decision_to_second_step_button = "MIV_LOYALTY_FACTORS_STEP"

		item = {
			value = miv_call_the_banners
			localization = interactive_bannermen_mode_decision
			current_description = interactive_bannermen_mode_decision_desc
			ai_chance = {
				value = 100
			}
		}
	}
	sort_order = 2001
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_destiny_goal.dds"
	}
}
