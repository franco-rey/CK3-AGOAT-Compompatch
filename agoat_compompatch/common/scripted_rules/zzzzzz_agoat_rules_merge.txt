can_be_knight = {
	trigger_if = {
		limit = {
			exists = liege
		}
		liege = { save_temporary_scope_as = army_owner }
	}
	trigger_else = {
		root.court_owner = { save_temporary_scope_as = army_owner }
	}

	trigger_if = {
		limit = {
			NOT = {
				exists = liege
			}
			OR = {
				is_playable_character = no
				root = root.court_owner
			}
		}
		trigger_if = {
			limit = {
				NOT = {
					has_game_rule = knight_age_disabled
				}
			}
			OR = {
				can_be_knight_trigger = {
					ARMY_OWNER = root.court_owner
				}
				miv_knight_age_override_trigger = {
					ARMY_OWNER = root.court_owner
				}
			}
		}
		trigger_else = {
			can_be_knight_trigger = {
				ARMY_OWNER = root.court_owner
			}
		}
	}
	trigger_else = {
		trigger_if = {
			limit = {
				NOT = {
					has_game_rule = knight_age_disabled
				}
			}
			OR = {
				can_be_knight_trigger = {
					ARMY_OWNER = liege
				}
				miv_knight_age_override_trigger = {
					ARMY_OWNER = liege
				}
				AND = {
					is_ai = no
					exists = liege
					OR = {
						is_adult = yes
						AND = {
							age >= 14
							is_current_dragonrider_warfare = yes
						}
					}
					is_clergy = no
					is_alive = yes
					is_imprisoned = no
					NOT = { has_trait = incapable }
					NOT = { has_trait = blind }
					NOT = { has_trait = devoted }
					NOT = { has_character_flag = cannot_be_knight_and_commander }
				}
			}
		}
		trigger_else = {
			OR = {
				can_be_knight_trigger = {
					ARMY_OWNER = liege
				}
				AND = {
					is_ai = no
					exists = liege
					OR = {
						is_adult = yes
						AND = {
							age >= 14
							is_current_dragonrider_warfare = yes
						}
					}
					is_clergy = no
					is_alive = yes
					is_imprisoned = no
					NOT = { has_trait = incapable }
					NOT = { has_trait = blind }
					NOT = { has_trait = devoted }
					NOT = { has_character_flag = cannot_be_knight_and_commander }
				}
			}
		}
		trigger_if = {
			limit = {
				has_game_rule = belligerence_miv_enabled
			}
			NOT = {
				liege = {
					has_opinion_modifier = {
						modifier = miv_oathbreaker
						target = prev
					}
				}
			}
		}
		trigger_if = {
			limit = {
				NOT = {
					has_game_rule = knight_eligibility_vassals_disabled
				}
			}
			trigger_if = {
				limit = {
					has_game_rule = knight_eligibility_vassals_enabled
				}
				NOT = {
					is_at_war = yes
				}
			}
			trigger_if = {
				limit = {
					has_game_rule = knight_eligibility_vassals_enabled_1
				}
				NOT = {
					highest_held_title_tier >= tier_kingdom
				}
			}
			trigger_if = {
				limit = {
					has_game_rule = knight_eligibility_vassals_enabled_2
				}
				NOT = {
					highest_held_title_tier >= tier_duchy
				}
			}
			trigger_if = {
				limit = {
					has_game_rule = knight_eligibility_vassals_enabled_3
				}
				NOT = {
					highest_held_title_tier >= tier_county
				}
			}
			trigger_if = {
				limit = {
					has_game_rule = knight_eligibility_vassals_enabled_4
				}
				NOT = {
					highest_held_title_tier >= tier_barony
				}
			}
		}
	}

	#AGOT Added
	NOT = { has_character_flag = agot_is_at_unknown_place }

	#AGOT Added, if commanding an army it should be owned by the same owner
	trigger_if = {
		limit = { exists = commanding_army }
		commanding_army.army_owner = scope:army_owner
	}
}

can_command_troops = {
	trigger_if = {
		limit = {
			NOT = {
				has_game_rule = commander_age_disabled
			}
		}
		OR = {
			can_be_commander_basic_trigger = {
				ARMY_OWNER = scope:army_owner
			}
			miv_commander_age_override_trigger = {
				ARMY_OWNER = scope:army_owner
			}
			AND = {
				is_ai = no
				OR = {
					is_adult = yes
					AND = {
						age >= 14
						is_current_dragonrider_warfare = yes
					}
				}
				is_clergy = no
				is_alive = yes
				is_imprisoned = no
				NOT = { has_trait = incapable }
				NOT = { has_trait = blind }
				NOT = { has_trait = devoted }
				OR = {
					NOT = { has_character_flag = cannot_be_knight_and_commander }
					scope:army_owner = this
				}
			}
		}
	}
	trigger_else = {
		OR = {
			can_be_commander_basic_trigger = {
				ARMY_OWNER = scope:army_owner
			}
			AND = {
				is_ai = no
				OR = {
					is_adult = yes
					AND = {
						age >= 14
						is_current_dragonrider_warfare = yes
					}
				}
				is_clergy = no
				is_alive = yes
				is_imprisoned = no
				NOT = { has_trait = incapable }
				NOT = { has_trait = blind }
				NOT = { has_trait = devoted }
				OR = {
					NOT = { has_character_flag = cannot_be_knight_and_commander }
					scope:army_owner = this
				}
			}
		}
	}
	trigger_if = {
		limit = {
			NOT = {
				has_game_rule = commander_eligibility_vassals_disabled
			}
		}
		trigger_if = {
			limit = {
				has_game_rule = commander_eligibility_vassals_enabled_0
			}
			OR = {
				is_courtier_of = scope:army_owner
				this = scope:army_owner
			}
		}
		trigger_if = {
			limit = {
				has_game_rule = commander_eligibility_vassals_enabled
			}
			OR = {
				dynasty = scope:army_owner.dynasty
				this = scope:army_owner
			}
		}
		trigger_if = {
			limit = {
				has_game_rule = commander_eligibility_vassals_enabled_2
			}
			OR = {
				house = scope:army_owner.house
				this = scope:army_owner
			}
		}
		trigger_if = {
			limit = {
				has_game_rule = commander_eligibility_vassals_enabled_3
			}
			OR = {
				is_close_or_extended_family_of = scope:army_owner
				this = scope:army_owner
			}
		}
		trigger_if = {
			limit = {
				has_game_rule = commander_eligibility_vassals_enabled_4
			}
			this = scope:army_owner
		}
	}
}

has_natural_death_second_chance = {
	OR = {
		AND = {
			has_perk = know_thyself_perk
			NOT = { has_character_flag = know_thyself_has_triggered_delay }
		}
		has_character_flag = know_thyself_perk_delay_period
		AND = {
			has_character_flag = eotog_warged
			NOT = { has_character_flag = know_thyself_has_triggered_delay }
		}
		AND = {
			NOT = { has_character_flag = ln_had_the_last_kiss }
			faith = { has_doctrine_parameter = can_heal_via_flames }
			ln_has_last_kiss_priest_trigger = yes
			ln_died_with_life_left_trigger = yes
			ln_worth_the_fire_trigger = yes
		}
	}
}

can_be_activity_guest = {
	ln_is_one_of_them_trigger = no
	is_available_allow_travelling = yes
	OR = {
		is_travelling = no
		# If travelling then it must be already to this activity after it has been created
		AND = {
			exists = scope:host.involved_activity
			exists = current_travel_plan.travel_plan_activity
			scope:host.involved_activity = current_travel_plan.travel_plan_activity
		}
	}
	# Diarchs with official responsibilities can't come out to play, unless within their liege's realm.
	trigger_if = {
		limit = {
			exists = liege
			OR = {
				is_diarch = yes
				is_designated_diarch = yes
			}
			#If you have promised a grand wedding, you must be able to go through with it!
			NOR = {
				has_promised_a_grand_wedding = yes
				has_been_promised_grand_wedding = yes
				liege = {
					has_diarchy_type = grand_secretariat
				}
			}
		}
		scope:host.involved_activity ?= {
			activity_location ?= {
				county ?= {
					holder ?= {
						OR = {
							this = root.liege
							any_liege_or_above = { this = root.liege }
						}
					}
				}
			}
		}
	}
	#AGOT Added
	trigger_if = {
		limit = {
			OR = {
				has_character_flag = exiled_from_iron_throne
				court_owner ?= {
					has_character_flag = exiled_from_iron_throne
				}
			}
		}
		scope:host.location.county.holder ?= {
			NOR = {
				has_title = title:h_the_iron_throne
				any_liege_or_above = { has_title = title:h_the_iron_throne }
				agot_is_part_of_paramountcy_realm = yes
			}
		}
	}
	#AGOT Added
	trigger_if = {
		limit = {
			scope:host ?= {
				has_character_flag = exiled_from_iron_throne
			}
		}
		NOR = {
			has_title = title:h_the_iron_throne
			any_liege_or_above = { has_title = title:h_the_iron_throne }
			agot_is_part_of_paramountcy_realm = yes
		}
	}
	trigger_if = {
		limit = {
			exists = house
			scope:host ?= {
				has_variable_list = guest_right_denied_houses
			}
		}
		custom_tooltip = {
			text = guest_right_house_denied_tt
			scope:host = {
				NOT = {
					is_target_in_variable_list = {
						name = guest_right_denied_houses
						target = root.house
					}
				}
			}
		}
	}
	trigger_if = {
		limit = {
			scope:host ?= {
				OR = {
					has_character_modifier = guest_right_breaker_modifier
					has_character_modifier = guest_right_captor_modifier
					has_character_modifier = guest_right_breaker_lenient_modifier
					has_character_modifier = guest_right_captor_lenient_modifier
				}
			}
		}
		custom_tooltip = {
			text = guest_right_host_is_breaker_tt
			OR = {
				this = scope:host
				court_owner ?= scope:host
			}
		}
	}
}
