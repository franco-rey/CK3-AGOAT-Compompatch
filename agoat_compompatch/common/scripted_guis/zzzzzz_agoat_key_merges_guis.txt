capture_wild_dragon = {
	scope = character

	is_shown = {
		scope:owner ?= {
			agot_has_an_active_dragonpit = yes
		}
		has_trait = dragon
		is_alive = yes
		OR = {
			any_liege_or_above = { this = scope:owner }
			location.county = {
				holder = {
					OR = {
						any_liege_or_above = { this = scope:owner }
						this = scope:owner
					}
				}
				agot_title_is_an_active_dragonpit = yes
			}
            scope:owner = {
                AND = {
                    is_landless_adventurer = yes
                    domicile ?= { domicile_location.county = root.location.county }
                }
            }
		}
		NOT = { has_character_flag = owned_dragon }
		NOT = { has_character_flag = in_dragonpit }
	}

	is_valid = {
		scope:owner = {
			custom_tooltip = {
				text = agot_capture_dragon_cd_gui_tt
				NOT = { has_variable = capture_cd }
			}
		}
		NOT = { lv_is_firewyrm = yes }
	}

	ai_is_valid = {
		scope:owner ?= { agot_has_an_active_dragonpit = yes }
		has_trait = dragon
		exists = employer
		NOT = { has_character_flag = owned_dragon }
		NOT = { has_character_flag = in_dragonpit }
		NOT = { lv_is_firewyrm = yes }
	}

	effect = {
		agot_change_dragonpit_status = yes
	}
}

can_reconstruct_ruin = {
	scope = province
	is_valid = {
		save_temporary_scope_as = ruin
		exists = scope:actor

		trigger_if = {
			limit = { lv_agot_ruin_uses_agot_restoration_trigger = yes }

			OR = {
				scope:actor = {
					any_sub_realm_county = {
						any_neighboring_county = { this = scope:ruin.county }
					}
				}
				scope:actor = {
					any_held_title = {
						tier > tier_county
						any_de_jure_county = { this = scope:ruin.county }
					}
				}
				scope:actor = {
					any_character_to_title_neighboring_and_across_water_county = {
						this = scope:ruin.county
					}
				}
				scope:actor = {
					domicile ?= {
						domicile_location.barony = {
							this = scope:ruin.barony
						}
					}
				}
				lv_agot_lov_special_access_trigger = { ACTOR = scope:actor }
			}

			trigger_if = {
				limit = { has_building = special_ruin_01 }
				lv_agot_ruin_can_pay_trigger = { AMOUNT = 1000 }
			}
			trigger_else_if = {
				limit = { has_building = large_ruin_03 }
				lv_agot_ruin_can_pay_trigger = { AMOUNT = 1800 }
			}
			trigger_else_if = {
				limit = { has_building = large_ruin_02 }
				lv_agot_ruin_can_pay_trigger = { AMOUNT = 1200 }
			}
			trigger_else_if = {
				limit = { has_building = large_ruin_01 }
				lv_agot_ruin_can_pay_trigger = { AMOUNT = 600 }
			}
			trigger_else_if = {
				limit = { has_building = medium_ruin_03 }
				lv_agot_ruin_can_pay_trigger = { AMOUNT = 1200 }
			}
			trigger_else_if = {
				limit = { has_building = medium_ruin_02 }
				lv_agot_ruin_can_pay_trigger = { AMOUNT = 800 }
			}
			trigger_else_if = {
				limit = { has_building = medium_ruin_01 }
				lv_agot_ruin_can_pay_trigger = { AMOUNT = 400 }
			}
			trigger_else_if = {
				limit = { has_building = small_ruin_03 }
				lv_agot_ruin_can_pay_trigger = { AMOUNT = 600 }
			}
			trigger_else_if = {
				limit = { has_building = small_ruin_02 }
				lv_agot_ruin_can_pay_trigger = { AMOUNT = 400 }
			}
			trigger_else_if = {
				limit = { has_building = small_ruin_01 }
				lv_agot_ruin_can_pay_trigger = { AMOUNT = 200 }
			}
			trigger_else_if = {
				limit = { has_building = jungle_ruin_deep_04 }
				lv_agot_ruin_can_pay_trigger = { AMOUNT = 1200 }
			}
			trigger_else_if = {
				limit = { has_building = jungle_ruin_deep_03 }
				lv_agot_ruin_can_pay_trigger = { AMOUNT = 800 }
			}
			trigger_else_if = {
				limit = { has_building = jungle_ruin_deep_02 }
				lv_agot_ruin_can_pay_trigger = { AMOUNT = 400 }
			}
			trigger_else_if = {
				limit = { has_building = jungle_ruin_deep_01 }
				lv_agot_ruin_can_pay_trigger = { AMOUNT = 400 }
			}
			trigger_else_if = {
				limit = { has_building = jungle_ruin_04 }
				lv_agot_ruin_can_pay_trigger = { AMOUNT = 600 }
			}
			trigger_else_if = {
				limit = { has_building = jungle_ruin_03 }
				lv_agot_ruin_can_pay_trigger = { AMOUNT = 400 }
			}
			trigger_else_if = {
				limit = { has_building = jungle_ruin_02 }
				lv_agot_ruin_can_pay_trigger = { AMOUNT = 200 }
			}
			trigger_else_if = {
				limit = { has_building = jungle_ruin_01 }
				lv_agot_ruin_can_pay_trigger = { AMOUNT = 200 }
			}
			trigger_else = { always = no }
		}
		trigger_else = {
			scope:actor = {
				gold >= lv_settlement_cost
				NOR = {
					var:lv_settlement_state ?= flag:planning
					any_owned_story = {
						story_type = lv_settlement_story_cycle
						is_target_in_variable_list = {
							name = lv_settlement_expeditions
							target = scope:ruin
						}
					}
				}
			}
		}
	}
}
