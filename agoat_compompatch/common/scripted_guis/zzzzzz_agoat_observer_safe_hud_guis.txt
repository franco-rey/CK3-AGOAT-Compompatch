# Several component HUDs continue evaluating hidden child widgets in observer mode.
# Their original character-scoped scripted GUIs therefore receive GetPlayer's
# invalid character scope every frame.  Rootless wrappers can safely return false
# when no player exists, while retaining the original single-player behaviour.

right_clear_notifications = {
	scope = none

	is_shown = {
		exists = scope:agoat_player
		scope:agoat_player ?= { exists = var:right_clear_notifications }
	}

	effect = {
		scope:agoat_player = {
			if = {
				limit = { exists = var:right_clear_notifications }
				remove_variable = right_clear_notifications
			}
			else = { set_variable = right_clear_notifications }
		}
	}
}

DI_hide_button_check = {
	scope = none

	is_valid = {
		exists = scope:agoat_player
		scope:agoat_player ?= {
			NOT = {
				any_owned_story = {
					story_type = DI_misc_variables_story
					has_variable = DI_button_invisible_var
				}
			}
		}
	}
}

DI_misc_variable_story_check = {
	scope = none

	is_valid = {
		exists = scope:agoat_player
		scope:agoat_player ?= {
			is_alive = yes
			OR = {
				NOT = { any_owned_story = { story_type = DI_misc_variables_story } }
				any_owned_story = {
					story_type = DI_misc_variables_story
					count > 1
				}
			}
		}
	}

	effect = {
		scope:agoat_player = {
			DI_clear_misc_variables_effect = yes
			if = {
				limit = { NOT = { any_owned_story = { story_type = DI_misc_variables_story } } }
				create_story = DI_misc_variables_story
			}
			else = {
				every_owned_story = {
					limit = { story_type = DI_misc_variables_story }
					end_story = yes
				}
			}
		}
	}
}
