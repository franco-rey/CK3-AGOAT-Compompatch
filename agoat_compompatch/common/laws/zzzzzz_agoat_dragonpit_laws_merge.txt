@dragonpit_law_cooldown_years = 5

dragonpit_laws = {
	can_change_law_group = { dragonlord_council_direct_law_change_allowed_trigger = yes }

	dragonpit_ruler_law = {
		flag = dragonpit_ruler
		can_have = {
			OR = {
				has_government = dragonlord_oligarchy_government
				any_held_title = {
					has_variable = has_dragonkeeper_order
				}
				domicile ?= { has_variable = has_dragonkeeper_order }
			}
		}
		can_pass = {
			dragonlord_council_direct_law_change_allowed_trigger = yes
			custom_description = {
				subject = root
				text = "has_dragonpit_law_cooldown"
				NOT = { has_variable = dragonpit_law_cooldown }
			}
		}

		pass_cost = {
			prestige = increase_crown_authority_prestige_cost
		}

		on_pass = {
			set_variable = {
				name = dragonpit_law_cooldown
				years = @dragonpit_law_cooldown_years
			}
		}

		can_keep = {
			OR = {
				has_government = dragonlord_oligarchy_government
				any_held_title = {
					has_variable = has_dragonkeeper_order
				}
				domicile ?= { has_variable = has_dragonkeeper_order }
			}
		}

		ai_will_do = {
			value = 0
		}
	}
