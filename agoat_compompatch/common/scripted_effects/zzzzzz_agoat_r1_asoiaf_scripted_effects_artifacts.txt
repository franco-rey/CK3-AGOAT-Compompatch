agot_create_artifact_robertI_crown_effect = {

	# Get the character the artifact is being made for.
	$OWNER$ = { save_scope_as = owner }
	$CREATOR$ = { save_scope_as = creator }

	set_artifact_rarity_illustrious = yes


	# Create the artifact
	if = {
		limit = {
			has_game_rule = asoiaf_new_clothes_stormlands_rule_on
		}
		create_artifact = {
			name = robertI_crown_name
			description = asoiaf_robert_I_crown_description #AGOT+ changed
			type = helmet
			visuals = asoiaf_robert_I_crown_visuals #AGOT+ changed
			wealth = scope:wealth
			quality = scope:quality
			template = robertI_crown_template
			modifier = robertI_crown_modifier
			save_scope_as = newly_created_artifact
			creator = scope:creator
			decaying = no
		}
	}
	else = {
		create_artifact = {
			name = robertI_crown_name
			description = robertI_crown_description
			type = helmet
			visuals = robertI_crown_visuals
			wealth = scope:wealth
			quality = scope:quality
			template = robertI_crown_template
			modifier = robertI_crown_modifier
			save_scope_as = newly_created_artifact
			creator = scope:creator
			decaying = no
		}
	}
	scope:newly_created_artifact ?= {
		set_variable = { name = robertI_crown_artifact value = yes }
		set_variable = { name = historical_unique_artifact value = yes }
		if = {
			limit = {
				scope:owner = { can_equip_artifact = prev }
			}
			equip_artifact_to_owner_replace = yes
		}
	}
}
