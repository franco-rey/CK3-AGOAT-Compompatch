naval_combat_shipwright_roll_quality_effect = {
	set_variable = { name = naval_combat_shipwright_discovery value = $DISCOVERY$ }
	if = { limit = { NOT = { exists = var:$UNLOCK_VARIABLE$ } } set_variable = { name = $UNLOCK_VARIABLE$ value = 0 } }

			set_variable = { name = naval_combat_shipwright_discovery_quality value = 1 }
		}
		40 = {
			trigger = { NOR = { var:$UNLOCK_VARIABLE$ ?= 2 var:$UNLOCK_VARIABLE$ ?= 3 var:$UNLOCK_VARIABLE$ ?= 6 var:$UNLOCK_VARIABLE$ ?= 7 var:$UNLOCK_VARIABLE$ ?= 10 var:$UNLOCK_VARIABLE$ ?= 11 var:$UNLOCK_VARIABLE$ ?= 14 var:$UNLOCK_VARIABLE$ ?= 15 } }

		}
	}
}
