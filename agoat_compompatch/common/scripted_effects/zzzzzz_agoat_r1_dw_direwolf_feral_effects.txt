dw_direwolf_go_feral_effect = {
    save_scope_as = dw_new_feral

    court_owner ?= { remove_courtier_or_guest = scope:dw_new_feral }

    scope:dw_new_feral = {
        remove_trait = direwolf_alpha
        remove_trait = direwolf_beta
        remove_trait = direwolf_omega
        remove_trait = direwolf_outcast
        remove_character_flag = dw_forced_outcast
        remove_character_flag = dw_pack_rep
        remove_trait = direwolf_bonded
        add_trait = direwolf_wild
        clear_variable_list = dw_packmates
        add_character_flag = dw_feral

        if = {

        	limit = { is_alive = no }

        	make_unprunable = yes

        }
    }

    add_to_global_variable_list = { name = dw_feral_wolves  target = scope:dw_new_feral }

    scope:dw_new_feral = { dw_direwolf_feral_join_or_create_effect = yes }
}

dw_direwolf_feral_leader_move_effect = {
    save_scope_as = dw_move_leader
    location.county ?= { save_scope_as = dw_move_old_county }

    if = {
        limit = { exists = location.county }
        location.county = {
            random_neighboring_county = {
                limit = {
                    trigger_if = {
                        limit = { exists = title_province }
                        title_province = {
                            NOT = { geographical_region = world_ruins }
                        }
                    }
                    trigger_else = { always = no }
                }
                save_scope_as = dw_move_new_county
            }
        }
    }

    if = {
        limit = { exists = scope:dw_move_new_county }
        scope:dw_move_leader = { set_location = scope:dw_move_new_county.title_province }
        every_in_global_list = {
            variable = dw_feral_wolves
            limit = {
                is_alive = yes
                NOT = { this = scope:dw_move_leader }
                var:dw_feral_leader = scope:dw_move_leader
            }
            set_location = scope:dw_move_new_county.title_province
        }
        scope:dw_move_new_county = { dw_direwolf_feral_recount_county_effect = yes }
        scope:dw_move_old_county ?= { dw_direwolf_feral_recount_county_effect = yes }

        every_player = {
            limit = { has_character_flag = dw_watch_feral_moves }
            send_interface_toast = {
                title = dw_direwolf_feral_pack_moved_toast
                left_icon = scope:dw_move_leader
            }
        }
    }
}

dw_direwolf_create_wild_wolf_effect = {
    create_character = {
        location = scope:dw_fpack_loc
        template = dw_direwolf_adult_template
        gender = $GENDER$
        random_traits = no
        save_scope_as = new_direwolf
    }
    scope:new_direwolf = {
        add_character_flag = dw_feral
        if = {
        	limit = { is_alive = no }
        	make_unprunable = yes
        }
    }
    add_to_global_variable_list = { name = dw_all_wolves     target = scope:new_direwolf }
    add_to_global_variable_list = { name = dw_feral_wolves   target = scope:new_direwolf }
    add_to_global_variable_list = { name = dw_spawning_pack  target = scope:new_direwolf }
}

dw_direwolf_spawn_feral_pack_beyond_wall_effect = {
    random_county = {
        limit = {
            trigger_if = {
                limit = { exists = title_province }
                title_province = {
                    geographical_region = world_westeros_beyond_the_wall
                }
            }
            trigger_else = { always = no }
        }
        save_scope_as = dw_btw_county
    }
    if = {
        limit = { exists = scope:dw_btw_county }
        scope:dw_btw_county.title_province = { save_scope_as = dw_fpack_loc }
        dw_direwolf_spawn_feral_pack_at_loc_effect = yes
    }
}
