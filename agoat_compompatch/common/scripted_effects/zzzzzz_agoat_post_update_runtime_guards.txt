	else_if = { limit = { has_character_flag = agot_guard_positions_rainbow_guard_member } create_character_memory = { type = agot_rainbow_guard_battle_victory_memory } }
	else_if = { limit = { has_character_flag = agot_guard_positions_battleguard_member } create_character_memory = { type = agot_battleguard_battle_victory_memory } }
	else_if = { limit = { has_character_flag = agot_guard_positions_winged_knight_member } create_character_memory = { type = agot_winged_knight_battle_victory_memory } }

	else_if = { limit = { has_character_flag = agot_guard_positions_greenhand_member } create_character_memory = { type = agot_greenhand_battle_victory_memory } }
	else_if = { limit = { has_character_flag = agot_guard_positions_tully_guard_member } create_character_memory = { type = agot_tully_guard_battle_victory_memory } }
	agot_guard_order_book_append_battle_effect = yes

		limit = { NOT = { has_variable = agot_guard_positions_service_battle_victories } }
		set_variable = { name = agot_guard_positions_service_battle_victories value = 0 }
	}
	change_variable = { name = agot_guard_positions_service_battle_victories add = 1 }
	agot_guard_positions_initialize_experience_effect = yes
	change_variable = { name = agot_guard_positions_battle_experience add = 1 }
	agot_guard_positions_update_specialisation_effect = yes
}

 if = {
  limit = { agot_guard_camp_selection_valid_trigger = yes exists = var:agot_guard_camp_selected_guard var:agot_guard_camp_selected_guard = { agot_guard_positions_guard_available_for_mission_trigger = { COURT_OWNER = root } agot_guard_camp_fighter_trigger = yes } }
  save_scope_as = camp_ruler
  set_variable = { name = agot_guard_camp_target value = var:agot_guard_camp_selected }
  set_variable = { name = agot_guard_camp_commander value = var:agot_guard_camp_selected_guard }
  var:agot_guard_camp_selected = {
   save_scope_as = camp_leader
   set_variable = { name = agot_guard_camp_targeted_by value = scope:camp_ruler }
   primary_title = { save_scope_as = camp_title }
   domicile.domicile_location = { save_scope_as = camp_location }
  }
  set_variable = { name = agot_guard_camp_title value = scope:camp_title }
 scope:camp_title = { set_variable = { name = agot_guard_camp_targeted_by value = scope:camp_ruler } }
  set_variable = { name = agot_guard_camp_location value = scope:camp_location }
  clear_variable_list = agot_guard_camp_members
  clear_variable_list = agot_guard_camp_guards
  add_to_variable_list = { name = agot_guard_camp_members target = scope:camp_leader }
  var:agot_guard_camp_selected = { every_courtier = { limit = { is_alive = yes is_human = yes } scope:camp_ruler = { add_to_variable_list = { name = agot_guard_camp_members target = prev } } } }
  add_to_variable_list = { name = agot_guard_camp_guards target = var:agot_guard_camp_selected_guard }
  var:agot_guard_camp_selected_guard = {
   add_character_flag = agot_guard_positions_on_mission
   add_character_modifier = agot_guard_positions_away_on_mission
   set_variable = { name = agot_guard_camp_mission_ruler value = scope:camp_ruler }
   set_location = scope:camp_location
  }
  agot_guard_camp_clear_selection_effect = yes
  trigger_event = { id = agot_guard_camps.0010 days = { 10 20 } }
 }
}

	set_variable = { name = agot_guard_house_lion_score value = 0 }

	set_variable = { name = agot_guard_house_falcon_score value = 0 }
	set_variable = { name = agot_guard_house_tower_score value = 0 }

	remove_character_flag = agot_guard_house_falcon_champion

		every_attending_character = {
			limit = { agot_guard_positions_is_training_for_active_order_trigger = { OWNER = root } }

	else = { add_character_flag = agot_guard_house_tower_champion }
	scope:activity = {
		ordered_attending_character = {
			position = 0
			limit = {
				agot_guard_positions_is_training_for_active_order_trigger = { OWNER = root }
				OR = {
					AND = { root = { has_character_flag = agot_guard_house_lion_champion } agot_guard_positions_academy_house_first_trigger = yes }
					AND = { root = { has_character_flag = agot_guard_house_stag_champion } agot_guard_positions_academy_house_second_trigger = yes }
					AND = { root = { has_character_flag = agot_guard_house_falcon_champion } agot_guard_positions_academy_house_third_trigger = yes }
					AND = { root = { has_character_flag = agot_guard_house_tower_champion } agot_guard_positions_academy_house_fourth_trigger = yes }
				}
			}
			order_by = var:agot_guard_positions_training_merit
			save_scope_as = academy_house_honours_captain
		}
	}
	scope:academy_house_honours_captain ?= { add_prestige = 75 }
	trigger_event = agot_guard_school_activity.0005
}

agot_guard_positions_resolve_national_team_effect = {
	scope:activity = {
		every_attending_character = {
			limit = { agot_guard_positions_is_actively_enrolled_in_any_academy_trigger = yes }
			var:agot_guard_positions_training_owner = { set_variable = { name = agot_guard_positions_national_team_score value = 0 } }
		}
		every_attending_character = {
			limit = { agot_guard_positions_is_actively_enrolled_in_any_academy_trigger = yes }
			var:agot_guard_positions_training_owner = {
				add_to_list = academy_national_team_holders
			}
			var:agot_guard_positions_training_owner = { change_variable = { name = agot_guard_positions_national_team_score add = prev.$PRIMARY$ } }
			var:agot_guard_positions_training_owner = { change_variable = { name = agot_guard_positions_national_team_score add = prev.$SECONDARY$ } }
		}
	}
	ordered_in_list = {
		list = academy_national_team_holders
		position = 0
		order_by = var:agot_guard_positions_national_team_score
		save_scope_as = academy_national_winning_school
	}
	scope:activity = {
		ordered_attending_character = {
			position = 0
			limit = { has_variable = agot_guard_positions_training_owner var:agot_guard_positions_training_owner ?= scope:academy_national_winning_school }
			order_by = { value = $PRIMARY$ add = $SECONDARY$ add = var:agot_guard_positions_training_merit }
			save_scope_as = academy_national_winner
		}
	}
	scope:academy_national_winning_school ?= {
		if = {
			limit = { NOT = { has_variable = agot_guard_positions_national_points } }
			set_variable = { name = agot_guard_positions_national_points value = 0 }
		}
		change_variable = { name = agot_guard_positions_national_points add = $POINTS$ }
		remove_variable = agot_guard_positions_national_team_score
	}
	scope:academy_national_winner ?= { agot_guard_positions_add_training_merit_effect = { AMOUNT = $MERIT$ } add_prestige = 50 agot_guard_positions_update_training_rank_effect = yes }
}
