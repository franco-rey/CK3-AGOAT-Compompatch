




ame_return_home_effect = {
	if = {
		limit = { is_imprisoned = yes }



		trigger_event = {
			on_action = ame_service_ongoing
			days = { 80 100 }
		}
	}
	else = {
		if = {
			limit = {
				exists = var:ame_home_liege
				var:ame_home_liege = { is_alive = yes }
			}
			var:ame_home_liege = {
				add_courtier = root
			}
		}
		else_if = {
			limit = {
				exists = dynasty.dynast
				dynasty.dynast = {
					OR = {
						is_landed = yes
						is_ruler = yes
					}
				}
			}
			dynasty.dynast = {
				add_courtier = root
			}
		}
		else_if = {


			limit = {
				exists = house.house_head
				house.house_head = {
					OR = {
						is_landed = yes
						is_ruler = yes
					}
				}
			}
			house.house_head = {
				add_courtier = root
			}
		}
		else = {
			move_to_pool = yes
		}




		if = {
			limit = { var:ame_years_of_service >= 2 }
			add_trait_force_tooltip = ame_mercenary_veteran
		}

		if = {
			limit = {
				exists = var:ame_home_liege
				var:ame_home_liege = { is_alive = yes }
			}


			save_scope_as = ame_returning_character
			var:ame_company = {
				save_scope_as = ame_returning_company
			}
			var:ame_home_liege = {
				trigger_event = ame_service.0011
			}
		}

		remove_character_flag = ame_serving
		remove_variable = ame_company
		remove_variable = ame_years_of_service
	}
}







ame_charge_service_cost_effect = {
	if = {
		limit = { NOT = { root = scope:recipient } }
		remove_short_term_gold = 75
	}
}






ame_join_company_effect = {
	title:$COMPANY$.holder = {
		add_courtier = $CHARACTER$
	}
	$CHARACTER$ = {
		set_variable = {
			name = ame_company
			value = title:$COMPANY$
		}


		set_variable = {
			name = ame_years_of_service
			value = 1
		}
		trigger_event = {
			on_action = ame_service_ongoing
			days = { 340 380 }
		}
	}
}






ame_revive_company_effect = {
	create_character = {
		age = 35
		gender = male
		location = title:$COMPANY$.title_capital_county.title_province
		culture = title:$COMPANY$.title_capital_county.title_province.culture
		faith = title:$COMPANY$.title_capital_county.title_province.faith
		save_scope_as = ame_new_captain
	}
	scope:ame_new_captain = {
		create_landless_adventurer_title_effect = { REASON = flag:voluntary FLAVOR_CHAR = scope:ame_new_captain }
	}
	create_title_and_vassal_change = { type = created save_scope_as = ame_revival_change add_claim_on_loss = no }
	title:$COMPANY$ = { change_title_holder = { holder = scope:ame_new_captain change = scope:ame_revival_change } }
	resolve_title_and_vassal_change = scope:ame_revival_change

	title:$COMPANY$.holder ?= {
		if = {
			limit = { NOT = { has_realm_law = camp_purpose_mercenaries } }
			add_realm_law = camp_purpose_mercenaries
		}
		hidden_effect = {
			create_maa_regiment = { type = light_footmen check_can_recruit = no size = 2 }
		}
	}


	title:$COMPANY$ = {
		set_variable = { name = ame_reform_cooldown value = yes years = 10 }
	}
}





















ame_veteran_backfill_check_effect = {
	$CHARACTER$ = {
		if = {
			limit = {



				is_male = yes
				NOT = { has_trait = ame_mercenary_veteran }


				NOT = { has_character_flag = ame_serving }
			}
			if = {


				limit = {
					exists = var:ame_veteran_backfill_tenure
					var:ame_veteran_backfill_company = title:$COMPANY$
				}
				add_trait_force_tooltip = ame_mercenary_veteran
				remove_variable = ame_veteran_backfill_tenure
				remove_variable = ame_veteran_backfill_company
			}
			else = {



				set_variable = { name = ame_veteran_backfill_tenure value = yes years = 2 }
				set_variable = { name = ame_veteran_backfill_company value = title:$COMPANY$ }
			}
		}
		else = {


			if = {
				limit = { exists = var:ame_veteran_backfill_tenure }
				remove_variable = ame_veteran_backfill_tenure
			}
			if = {
				limit = { exists = var:ame_veteran_backfill_company }
				remove_variable = ame_veteran_backfill_company
			}
		}
	}
}








ame_veteran_backfill_apply_to_company_effect = {
	if = {
		limit = { exists = title:$COMPANY$.holder }
		ame_veteran_backfill_check_effect = { CHARACTER = title:$COMPANY$.holder COMPANY = $COMPANY$ }
		title:$COMPANY$.holder = {
			every_courtier = {
				limit = { is_adult = yes }
				ame_veteran_backfill_check_effect = { CHARACTER = this COMPANY = $COMPANY$ }
			}
		}
	}
}
