portrait_wear_helmet_trigger = {
	exists = this
	trigger_if = {
		limit = { exists = involved_activity }
		involved_activity = {
			NOR = {
				has_activity_type = activity_hunt
				has_activity_type = activity_pilgrimage
			}
		}
	}
	OR = {

		AND = {
			is_ruler = yes
			government_has_flag = government_is_mercenary
			is_at_war = yes
		}

		AND = {
			is_ruler = yes
			government_has_flag = government_is_holy_order
			is_at_war = yes
		}
		AND = {
			has_trait = order_member
			is_landed = no
			exists = liege
			liege = { is_at_war = yes }
		}


			is_commanding_army = yes


		exists = knight_army


		has_character_flag = wear_armor

		has_character_flag = is_currently_varangian

		has_character_flag = single_combat_duel_armor

		activity_tournament_armor_trigger = yes


	}
	NOT = { any_equipped_character_artifact = { artifact_type = helmet } }
	NOR = {
		activity_tournament_shirtless_trigger = yes
		has_character_flag = no_hat
	}

	NAND = {
		has_trait = peasant_leader
		OR = {
			is_ruler = no
			is_leading_faction_type = peasant_faction
			AND = {
				is_ruler = yes
				is_landed = no
				OR = {
					government_has_flag = government_is_feudal
					government_has_flag = government_is_clan
				}
			}
		}
	}
}

portrait_wear_lannister_armour_courtiers_trigger = {
	exists = this
	trigger_if = {
		limit = { exists = involved_activity }
		involved_activity = {
			NOR = {
				has_activity_type = activity_hunt
				has_activity_type = activity_pilgrimage
			}
		}
	}
	is_male = yes
	is_courtier = yes
	is_foreign_court_guest = no
	OR = {
		this.liege.dynasty ?= dynasty:dynn_Lannister
		this.liege.house ?= house:house_BaratheonKL
	}
	OR = {
		this.liege.culture ?= culture:westerman_main
		this.liege.culture ?= culture:crownlander
		this.liege.culture ?= culture:reachman
		this.liege.culture ?= culture:riverman_main
		this.liege.culture ?= culture:valeman_main
		this.liege.culture ?= culture:stormlander_main
	}
	culture = { has_cultural_pillar = heritage_andal }
	portrait_wear_armor_trigger = yes
	NOT = { has_trait = peasant_leader }
	NOR = {
		has_trait = peasant_leader
		has_trait = maester
	}
	has_game_rule = asoiaf_new_clothes_westerlands_rule_on
}
