agot_flavour_is_in_westeros_trigger = {
	trigger_if = {
		limit = { $CHAR$ ?= { is_alive = no } }
		$CHAR$ ?= {
			primary_title ?= {
				title_province ?= { geographical_region = world_westeros_seven_kingdoms_without_wall }
			}
		}
	}
	trigger_else = {
		agot_flavour_is_living_in_westerosi_realm_trigger = { CHAR = $CHAR$ }
	}
}

agot_flavour_is_living_in_westerosi_realm_trigger = {
	$CHAR$ ?= {
		is_alive = yes
		is_landed = yes

		# Baron > Count Liege
		trigger_if = {
			limit = { exists = top_liege }
				trigger_if = {
					limit = {
					top_liege ?= { has_variable = agot_pwl_direct }
					}
				top_liege.var:agot_pwl_direct ?= {

					trigger_if = {
						limit = { exists = top_liege }
						trigger_if = {
							limit = {
								top_liege ?= { has_variable = agot_pwl_direct }
							}
							top_liege.var:agot_pwl_direct ?= {

								trigger_if = {
									limit = { exists = top_liege }
								trigger_if = {
									limit = {
											top_liege ?= { has_variable = agot_pwl_direct }
									}
										top_liege.var:agot_pwl_direct ?= {

											trigger_if = {
												limit = { exists = top_liege }
										trigger_if = {
											limit = {
														top_liege ?= { has_variable = agot_pwl_direct }
													}
													top_liege.var:agot_pwl_direct ?= {

														trigger_if = {
															limit = { exists = top_liege }
															trigger_if = {
																limit = {
																	top_liege ?= { has_variable = agot_pwl_direct }
											}
										}
										trigger_else = {
																top_liege ?= { agot_flavour_capital_is_in_westerosi_trigger = yes }
										}
														}
														trigger_else = { always = no }
									}
								}
								trigger_else = {
													top_liege ?= { agot_flavour_capital_is_in_westerosi_trigger = yes }
								}
											}
											trigger_else = { always = no }
							}
						}
						trigger_else = {
										top_liege ?= { agot_flavour_capital_is_in_westerosi_trigger = yes }
						}
								}
								trigger_else = { always = no }
					}
				}
				trigger_else = {
							top_liege ?= { agot_flavour_capital_is_in_westerosi_trigger = yes }
				}
					}
					trigger_else = { always = no }
			}
		}
		trigger_else = {
				top_liege ?= { agot_flavour_capital_is_in_westerosi_trigger = yes }
		}
		}
		trigger_else = { always = no }
	}
}

agot_flavour_is_lord_in_westeros_trigger = {
	$CHAR$ ?= {
		trigger_if = { # if char is dead check inactive trait
			limit = { is_alive = no }
			has_inactive_trait = agot_was_westerosi_lord
		}
		trigger_else = { # else check dynamic requirements
			agot_flavour_is_living_in_westerosi_realm_trigger = { CHAR = this }
			agot_is_independent_ruler = no
		}
	}
}
