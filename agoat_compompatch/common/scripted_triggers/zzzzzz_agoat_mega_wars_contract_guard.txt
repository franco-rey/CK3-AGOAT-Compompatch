agot_mw_uses_feudal_government_taxes_trigger = {
	trigger_if = {
		limit = { exists = vassal_contract_obligation_level:feudal_government_taxes }
		OR = {
			subject_contract_obligation_level_can_be_increased = feudal_government_taxes
			subject_contract_obligation_level_can_be_decreased = feudal_government_taxes
		}
	}
	trigger_else = { always = no }
}
