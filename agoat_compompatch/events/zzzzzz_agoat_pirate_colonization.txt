namespace = agoat_pirate_colonization

agoat_pirate_colonization.9200 = {
	hidden = yes

	trigger = {
		NOT = { has_character_flag = recently_compensated }
		NOT = { government_has_flag = government_is_pirate_trigger_check }
		any_held_title = {
			tier = tier_county
			title_province = {
				geographical_region = historical_pirate_region
				OR = {
					has_holding_type = pirate_den_holding
					AND = {
						has_holding_type = settlement_holding
						has_building = pirate_remnants
					}
				}
			}
		}
	}

	immediate = {
		if = {
			limit = {
				is_independent_ruler = no
				liege ?= { is_ai = no }
			}
			if = {
				limit = { gold < 500 }
				save_scope_as = colonist
				liege = { trigger_event = agot_pirate_events.9201 }
			}
		}
		else_if = {
			limit = {
				any_held_county = {
					title_province = {
						geographical_region = historical_pirate_region
						has_holding_type = pirate_den_holding
					}
					count = all
				}
			}

			if = {
				limit = { is_independent_ruler = no }

				create_title_and_vassal_change = {
					type = independency
					save_scope_as = independence_change
					add_claim_on_loss = no
				}
				becomes_independent = {
					change = scope:independence_change
				}
				resolve_title_and_vassal_change = scope:independence_change
			}

			if = {
				limit = {
					agot_dlc_check_for_landless_pirates = yes
				}
				root = { change_government = pirate_government }
			}
			else = {
				root = { change_government = pirate_no_dlc_government }
			}
		}
		else = {
			every_held_title = {
				limit = {
					tier = tier_county
					title_province = {
						geographical_region = historical_pirate_region
						OR = {
							has_holding_type = pirate_den_holding
							AND = {
								has_holding_type = settlement_holding
								has_building = pirate_remnants
							}
						}
					}
				}
				save_scope_as = current_county
				make_settlement_county_wilderness = { COUNTY = scope:current_county }
			}
		}
	}
}
