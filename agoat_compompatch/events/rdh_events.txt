revive_dead_house.01 = {
	type = character_event
	title = revive_dead_house.01.t
	desc = revive_dead_house.01.desc
	theme = dynasty
	left_portrait = {
		character = root
		animation = shock
	}
	lower_left_portrait = {
		character = scope:gui_founder
	}

	option = {
		name = revive_dead_house.01.a

		# 你好
		save_scope_value_as = { name = rdh_renown_loss value = rdh_value_renown_loss }
		dynasty = { add_dynasty_prestige = scope:rdh_renown_loss }

		trigger_event = {
			id = revive_dead_house.02
		}
	}

	option = {
		name = revive_dead_house.01.b
		if = {
			limit = {
				debug_only = no
				has_character_flag = found_members_from_dead_house_recently
			}
			remove_character_flag = found_members_from_dead_house_recently
		}
	}
}
