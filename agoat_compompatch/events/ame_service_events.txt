			triggered_desc = {
				trigger = {
					scope:ame_returning_character = {
						OR = {
							has_trait = ame_haunted_by_war_1
							has_trait = ame_haunted_by_war_2
						}
					}
				}
				desc = ame_service.0011.desc_haunted
			}
			triggered_desc = {
				trigger = {
					scope:ame_returning_character = {
						OR = {
							has_trait = ame_tempered_by_war_1
							has_trait = ame_tempered_by_war_2
						}
					}
				}
				desc = ame_service.0011.desc_tempered
			}
			triggered_desc = {
				trigger = {
					scope:ame_returning_character = { var:ame_years_of_service >= 5 }
				}
				desc = ame_service.0011.desc_long_service
			}
			# Fires under 2 years served, before desc_simple's "thinner and quieter" framing would fit.

				trigger = {
					scope:ame_returning_character = { var:ame_years_of_service < 2 }
				}
				desc = ame_service.0011.desc_early_recall
			}
			desc = ame_service.0011.desc_simple
		}
	}
	sender = scope:ame_returning_character

	option = {
		name = ame_service.0011.a
	}
}

# Death in service. Drawn from ame_service_ongoing's random_events. weight_multiplier scales the
# baseline weight up while the company is at war (the "real teeth" half of the mixture).
ame_service.0020 = {
	type = character_event
	title = ame_service.0020.t
	desc = ame_service.0020.desc
	theme = war

	left_portrait = root
	right_portrait = var:ame_company.holder

	trigger = {
		has_character_flag = ame_serving
	}

	weight_multiplier = {
		base = 1
		modifier = {
			var:ame_company.holder ?= { is_at_war = yes }
			factor = 4
		}
		# Half weight for a son serving in a company his home liege owns outright (war modifier above).
		# See ame_serving_under_own_patron_trigger for what "owns outright" means.
		modifier = {
			ame_serving_under_own_patron_trigger = yes
			factor = 0.5
		}
	}

	option = {
		name = ame_service.0020.a

		if = {
			limit = {
				exists = var:ame_home_liege
				var:ame_home_liege ?= { is_alive = yes }
				exists = var:ame_company.holder
				var:ame_company.holder ?= { is_alive = yes }
			}
			save_scope_as = ame_fallen_character
			var:ame_company = {
				save_scope_as = ame_fallen_company
			}
			var:ame_company.holder = {
				save_scope_as = ame_reporting_captain
			}
			save_scope_value_as = { name = agoat_ame_fallen_service_years value = 0 }
			if = {
				limit = { has_variable = ame_years_of_service }
				save_scope_value_as = { name = agoat_ame_fallen_service_years value = var:ame_years_of_service }
			}
			var:ame_home_liege = {
				save_scope_as = agoat_ame_fallen_home_liege
			}
		}

		if = {
			limit = { has_character_flag = ame_serving }
			remove_character_flag = ame_serving
		}
		if = {
			limit = { has_variable = ame_company }
			remove_variable = ame_company
		}
		if = {
			limit = { has_variable = ame_years_of_service }
			remove_variable = ame_years_of_service
		}
		death = { death_reason = death_battle }
		scope:agoat_ame_fallen_home_liege ?= {
			if = {
				limit = { is_alive = yes }
				trigger_event = ame_service.0021
			}
		}

	}
}

# Horrors of war. Mutually exclusive with ame_tempered_by_war (cross-guard below): tempered stays tempered
# and just pays the stress cost; otherwise escalates ame_haunted_by_war tier 1->2, capped.
ame_service.0030 = {
	type = character_event
	title = ame_service.0030.t
	desc = ame_service.0030.desc
	theme = mental_health

	left_portrait = root

	trigger = {
		has_character_flag = ame_serving
	}

	weight_multiplier = {
		base = 1
		modifier = {
			var:ame_company.holder ?= { is_at_war = yes }
			factor = 3
		}
		# Same modifier as ame_service.0020's weight_multiplier: half weight for a son serving in a
		# company his home liege owns outright.
		modifier = {
			ame_serving_under_own_patron_trigger = yes
			factor = 0.5
		}
	}

	immediate = {
		if = {
			limit = {
				NOR = {
					has_trait = ame_tempered_by_war_1
					has_trait = ame_tempered_by_war_2
				}
			}
			if = {
				limit = { has_trait = ame_haunted_by_war_1 }
				change_trait_rank = {
					trait = ame_haunted_by_war
					rank = 2
					max = 2
				}
			}
			else_if = {
				limit = { has_trait = ame_haunted_by_war_2 }
				# Already at the worst tier. No further escalation, just the stress below.
			}
			else = {
				change_trait_rank = {
					trait = ame_haunted_by_war
					rank = 1
					max = 2
				}
				create_character_memory = {
					type = ame_haunted_by_war_memory
				}
			}
		}
		add_stress = major_stress_gain
	}

	option = {
		name = ame_service.0030.a
	}
}

			}
			25 = {
				set_local_variable = { name = ame_reward_kind value = 4 }
				random_list = {
					25 = { add_martial_skill = 1 }
					25 = { add_prowess_skill = 1 }
					25 = { add_diplomacy_skill = 1 }
					25 = { add_intrigue_skill = 1 }
				}
			}
		}
		# Flat top-up on whatever the roll above gave, for a son serving in a company his home liege owns
		# outright - same rule as ame_service.0020's weight_multiplier.
		if = {
			limit = { ame_serving_under_own_patron_trigger = yes }
			add_gold = minor_gold_value
		}
	}

	option = {
		name = ame_service.0040.a
	}
}

# Tempered by war, the positive mirror of 0030 (same trigger, opposite outcome). Mutually exclusive
# with ame_haunted_by_war: haunted stays haunted and just gets the stress relief below.
ame_service.0060 = {
	type = character_event
	title = ame_service.0060.t
	desc = ame_service.0060.desc
	theme = martial

	left_portrait = root

	trigger = {
		has_character_flag = ame_serving
	}

	weight_multiplier = {
		base = 1
		modifier = {
			var:ame_company.holder ?= { is_at_war = yes }
			factor = 3
		}
	}

	immediate = {
		if = {
			limit = {
				NOR = {
					has_trait = ame_haunted_by_war_1
					has_trait = ame_haunted_by_war_2
				}
			}
			if = {
				limit = { has_trait = ame_tempered_by_war_1 }
				change_trait_rank = {
					trait = ame_tempered_by_war
					rank = 2
					max = 2
				}
			}
			else_if = {
				limit = { has_trait = ame_tempered_by_war_2 }
				# Already at the best tier. No further escalation, just the stress relief below.
			}
			else = {
				change_trait_rank = {
					trait = ame_tempered_by_war
					rank = 1
					max = 2
				}
				create_character_memory = {
					type = ame_tempered_by_war_memory
				}
			}
		}
		add_stress = minor_stress_loss
	}

	option = {
		name = ame_service.0060.a
	}
}
