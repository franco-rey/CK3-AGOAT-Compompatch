ame_service.0011 = {
	type = letter_event
	opening = ame_service.0011.opening
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:ame_returning_character = {
						OR = {
							has_trait = ame_haunted_by_war_1
							has_trait = ame_haunted_by_war_2
						}
					}
				}
				desc = ame_service.0011.desc_haunted
			}
			triggered_desc = {
				trigger = {
					scope:ame_returning_character = {
						OR = {
							has_trait = ame_tempered_by_war_1
							has_trait = ame_tempered_by_war_2
						}
					}
				}
				desc = ame_service.0011.desc_tempered
			}
			triggered_desc = {
				trigger = {
					scope:ame_returning_character = { var:ame_years_of_service >= 5 }
				}
				desc = ame_service.0011.desc_long_service
			}



			triggered_desc = {
				trigger = {
					scope:ame_returning_character = { var:ame_years_of_service < 2 }
				}
				desc = ame_service.0011.desc_early_recall
			}
			desc = ame_service.0011.desc_simple
		}
	}
	sender = scope:ame_returning_character

	option = {
		name = ame_service.0011.a
	}

ame_service.0020 = {
	type = character_event
	title = ame_service.0020.t
	desc = ame_service.0020.desc
	theme = war

	left_portrait = root
	right_portrait = var:ame_company.holder

	trigger = {
		has_character_flag = ame_serving
	}

	weight_multiplier = {
		base = 1
		modifier = {
			var:ame_company.holder = { is_at_war = yes }
			factor = 4
		}
	}

	option = {
		name = ame_service.0020.a

		death = { death_reason = death_battle }

		if = {
			limit = {
				exists = var:ame_home_liege
				var:ame_home_liege = { is_alive = yes }
				exists = var:ame_company.holder
				var:ame_company.holder = { is_alive = yes }
			}


			save_scope_as = ame_fallen_character
			var:ame_company = {
				save_scope_as = ame_fallen_company
			}
			var:ame_company.holder = {
				save_scope_as = ame_reporting_captain
			}
			var:ame_home_liege = {
				trigger_event = ame_service.0021
			}
		}

		if = {
			limit = { has_character_flag = ame_serving }
			remove_character_flag = ame_serving
		}
		if = {
			limit = { has_variable = ame_company }
			remove_variable = ame_company
		}
		if = {
			limit = { has_variable = ame_years_of_service }
			remove_variable = ame_years_of_service
		}
	}
}

ame_service.0030 = {
	type = character_event
	title = ame_service.0030.t
	desc = ame_service.0030.desc
	theme = mental_health

	left_portrait = root

	trigger = {
		has_character_flag = ame_serving
	}

	weight_multiplier = {
		base = 1
		modifier = {
			var:ame_company.holder = { is_at_war = yes }
			factor = 3
		}
	}

	immediate = {
		if = {
			limit = {
				NOR = {
					has_trait = ame_tempered_by_war_1
					has_trait = ame_tempered_by_war_2
				}
			}
			if = {
				limit = { has_trait = ame_haunted_by_war_1 }
				change_trait_rank = {
					trait = ame_haunted_by_war
					rank = 2
					max = 2
				}
			}
			else_if = {
				limit = { has_trait = ame_haunted_by_war_2 }

			}
			else = {
				change_trait_rank = {
					trait = ame_haunted_by_war
					rank = 1
					max = 2
				}
				create_character_memory = {
					type = ame_haunted_by_war_memory
				}
			}
		}
		add_stress = major_stress_gain
	}

	option = {
		name = ame_service.0030.a
	}
}



ame_service.0040 = {
	type = character_event
	title = ame_service.0040.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { local_var:ame_reward_kind = 1 }
				desc = ame_service.0040.desc_gold
			}
			triggered_desc = {
				trigger = { local_var:ame_reward_kind = 2 }
				desc = ame_service.0040.desc_prestige
			}
			triggered_desc = {
				trigger = { local_var:ame_reward_kind = 3 }
				desc = ame_service.0040.desc_camaraderie
			}
			desc = ame_service.0040.desc_skill
		}
	}
	theme = martial

	left_portrait = root

	trigger = {
		has_character_flag = ame_serving
	}

	immediate = {
		random_list = {
			30 = {
				set_local_variable = { name = ame_reward_kind value = 1 }
				add_gold = medium_gold_value
			}
			25 = {
				set_local_variable = { name = ame_reward_kind value = 2 }
				add_prestige = medium_prestige_gain
			}
			20 = {
				set_local_variable = { name = ame_reward_kind value = 3 }
				add_stress = minor_stress_loss
			}
			25 = {
				set_local_variable = { name = ame_reward_kind value = 4 }
				random_list = {
					25 = { add_martial_skill = 1 }
					25 = { add_prowess_skill = 1 }
					25 = { add_diplomacy_skill = 1 }
					25 = { add_intrigue_skill = 1 }
				}
			}
		}
	}

	option = {
		name = ame_service.0040.a
	}
