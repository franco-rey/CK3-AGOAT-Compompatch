agot_on_change_government_data_effect = {
	### Set Government ###
	$SCOPE$ ?= {
		switch = {
			trigger = has_government

			lp_feudal_government = { assign_government_data_effect = { GOV = lp_feudal SCOPE = $SCOPE$ } }
			feudal_government = { assign_government_data_effect = { GOV = feudal SCOPE = $SCOPE$ } }
			administrative_government = { assign_government_data_effect = { GOV = administrative SCOPE = $SCOPE$ } }
			norvos_government = { assign_government_data_effect = { GOV = norvos SCOPE = $SCOPE$ } }
			lorathi_principality_government = { assign_government_data_effect = { GOV = lorathi_principality SCOPE = $SCOPE$ } }
			republic_government = { assign_government_data_effect = { GOV = republic SCOPE = $SCOPE$ } }
			oligarchic_government = { assign_government_data_effect = { GOV = oligarchic SCOPE = $SCOPE$ } }
			silent_sisterhood_government = { assign_government_data_effect = { GOV = silent_sisterhood SCOPE = $SCOPE$ } }
			kingsguard_government = { assign_government_data_effect = { GOV = kingsguard SCOPE = $SCOPE$ } }
			nights_watch_government = { assign_government_data_effect = { GOV = nights_watch SCOPE = $SCOPE$ } }
			first_ranger_government = { assign_government_data_effect = { GOV = first_ranger SCOPE = $SCOPE$ } }
			the_citadel_government = { assign_government_data_effect = { GOV = the_citadel SCOPE = $SCOPE$ } }
			theocracy_government = { assign_government_data_effect = { GOV = theocracy SCOPE = $SCOPE$ } }
			pirate_government = { assign_government_data_effect = { GOV = pirate SCOPE = $SCOPE$ } }
			pirate_no_dlc_government = { assign_government_data_effect = { GOV = pirate_no_dlc SCOPE = $SCOPE$ } }
			tribal_government = { assign_government_data_effect = { GOV = tribal SCOPE = $SCOPE$ } }
			clan_government = { assign_government_data_effect = { GOV = clan SCOPE = $SCOPE$ } }
			landless_adventurer_government = { assign_government_data_effect = { GOV = landless_adventurer SCOPE = $SCOPE$ } }
			command_government = { assign_government_data_effect = { GOV = command SCOPE = $SCOPE$ } }
			ruins_government = { assign_government_data_effect = { GOV = ruins SCOPE = $SCOPE$ } }
			fallback = { prune_government_data_effect = { SCOPE = $SCOPE$ } }
		}
		if = {
			limit = { has_variable = temp_government_type }
			remove_variable = temp_government_type
		}
	}
}

assign_government_data_effect = {
	### REMOVE FROM OLD LISTS ###
	if = {
		limit = { NOT = { flag:$GOV$ = flag:lp_feudal } }
		remove_list_global_variable = { name = lp_feudal_government_list target = $SCOPE$ }
	}
	if = {
		limit = { NOT = { flag:$GOV$ = flag:feudal } }
		remove_list_global_variable = { name = feudal_government_list target = $SCOPE$ }
	}
	if = {
		limit = { NOT = { flag:$GOV$ = flag:administrative } }
		remove_list_global_variable = { name = administrative_government_list target = $SCOPE$ }
	}
	if = {
		limit = { NOT = { flag:$GOV$ = flag:norvos } }
		remove_list_global_variable = { name = norvos_government_list target = $SCOPE$ }
	}
	if = {

	}
	if = {
		limit = { NOT = { flag:$GOV$ = flag:republic } }

	}
	if = {
		limit = { NOT = { flag:$GOV$ = flag:oligarchic } }

	if = {

	}
	if = {
		limit = { NOT = { flag:$GOV$ = flag:nights_watch } }
		remove_list_global_variable = { name = nights_watch_government_list target = $SCOPE$ }
	}
	if = {
		limit = { NOT = { flag:$GOV$ = flag:first_ranger } }
		remove_list_global_variable = { name = first_ranger_government_list target = $SCOPE$ }
	}
	if = {
		limit = { NOT = { flag:$GOV$ = flag:the_citadel } }
		remove_list_global_variable = { name = the_citadel_government_list target = $SCOPE$ }
	}
	if = {
		limit = { NOT = { flag:$GOV$ = flag:theocracy } }
		remove_list_global_variable = { name = theocracy_government_list target = $SCOPE$ }
	}
	if = {
		limit = { NOT = { flag:$GOV$ = flag:pirate } }
		remove_list_global_variable = { name = pirate_government_list target = $SCOPE$ }
	}
	if = {
		limit = { NOT = { flag:$GOV$ = flag:pirate_no_dlc } }
		remove_list_global_variable = { name = pirate_no_dlc_government_list target = $SCOPE$ }
	}
	if = {
		limit = { NOT = { flag:$GOV$ = flag:tribal } }
		remove_list_global_variable = { name = tribal_government_list target = $SCOPE$ }
	}
	if = {

	if = {
		limit = { NOT = { flag:$GOV$ = flag:landless_adventurer } }
		remove_list_global_variable = { name = landless_adventurer_government_list target = $SCOPE$ }
	}
	if = {
		limit = { NOT = { flag:$GOV$ = flag:command } }
		remove_list_global_variable = { name = command_government_list target = $SCOPE$ }
	}
	if = {
		limit = { NOT = { flag:$GOV$ = flag:ruins } }
		remove_list_global_variable = { name = ruins_government_list target = $SCOPE$ }
	}

	### ASSIGN TO LIST ###
	if = {
		limit = {
			NOT = { is_target_in_global_variable_list = { name = $GOV$_government_list target = $SCOPE$ } }
		}
		add_to_global_variable_list = {
			name = $GOV$_government_list
			target = $SCOPE$
		}
	}
}
