# TITLE TRACKING (Creation + Conquest)
# HOTFIX 1.8.1: NO trigger block here. Per _on_actions.info an on_action can
# have only ONE trigger across all files; vanilla's on_title_gain has none,
# so a trigger added here became the trigger for the MERGED on_action and
# silently gated vanilla's own county/nomad title-handling effect to players
# only (broke AI nomad titles). The player gate lives in gr_tracking.1's own
# event trigger instead, and gr_internal_title_story_tracking is gated
# internally. RULE: never add trigger/effect blocks to a shared on_action.
# ================================================
on_title_gain = {
	events = {
		gr_tracking.1
	}
	on_actions = {
		gr_internal_title_story_tracking
	}
}

# Story panel increment for title gains (duchy+ only)
gr_internal_title_story_tracking = {
	effect = {
		if = {
			limit = {
				is_ai = no
				exists = scope:title
				scope:title = { tier >= tier_duchy }
				NOT = { has_variable = gr_succession_lock }
			}
			gr_story_increment_event = { TYPE = flag:title }
		}
	}
}

# ================================================
# BATTLE TRACKING
# Uses nested on_action to avoid conflicting with vanilla
# Root = Winning combat side
# ================================================
on_combat_end_winner = {
	on_actions = {
		gr_internal_battle_tracking
	}
}

gr_internal_battle_tracking = {
	effect = {
		# Check if player is the commander
		if = {
			limit = {
				exists = side_commander
				side_commander = { gr_is_tracked_character = yes }
			}
			
			# Get enemy casualties
			if = {
				limit = {
					exists = enemy_side
				}
				
				# Calculate enemy casualties (max - current = killed)
				save_temporary_scope_value_as = {
					name = enemy_max
					value = enemy_side.side_max_army_size
				}
				save_temporary_scope_value_as = {
					name = enemy_current
					value = enemy_side.side_army_size
				}
				save_temporary_scope_value_as = {
					name = battle_casualties
					value = {
						value = scope:enemy_max
						subtract = scope:enemy_current
					}
				}
				
				# Apply to the player commander
				side_commander = {
					# Mark that they led troops personally
					if = {
						limit = { NOT = { has_variable = gr_memory_led_troops } }
						set_variable = { name = gr_memory_led_troops value = yes }
					}
					
					# Add to cumulative casualty count
					if = {
						limit = { NOT = { has_variable = gr_total_casualties } }
						set_variable = { name = gr_total_casualties value = 0 }
					}
					change_variable = {
						name = gr_total_casualties
						add = scope:battle_casualties
					}
					
					# Track significant battles (1000+ enemy troops)
					if = {
						limit = { scope:enemy_max >= 1000 }
						
						# Only save if better than previous best
						if = {
							limit = {
								OR = {
									NOT = { has_variable = gr_best_battle_strength }
									scope:enemy_max > var:gr_best_battle_strength
								}
							}
							
							set_variable = {
								name = gr_best_battle_strength
								value = scope:enemy_max
							}
							
							set_variable = {
								name = gr_memory_great_battle
								value = yes
							}
							
							# Determine battle tier
							if = {
								limit = { scope:enemy_max >= 10000 }
								set_variable = { name = gr_battle_tier value = flag:legendary }
							}
							else_if = {
								limit = { scope:enemy_max >= 5000 }
								set_variable = { name = gr_battle_tier value = flag:massive }
							}
							else_if = {
								limit = { scope:enemy_max >= 3000 }
								set_variable = { name = gr_battle_tier value = flag:major }
							}
							else = {
								set_variable = { name = gr_battle_tier value = flag:significant }
							}
						}
					}
				}
			}
		}
	}
}

# WAR WON TRACKING (Varangian + other)
# Uses nested on_action to avoid overriding vanilla's
# effect block (which handles Temujin rewards, legend
# seeds, war notifications, etc.)
# ================================================
on_war_won_attacker = {
	on_actions = {
		gr_internal_war_won_tracking
	}
}

gr_internal_war_won_tracking = {
	effect = {
		if = {
			limit = {
				exists = scope:attacker
				scope:attacker = { gr_is_tracked_character = yes }
			}
			
			# Track that player won any war
			scope:attacker = {
				set_variable = { name = gr_memory_won_war value = yes }
				set_variable = { name = gr_memory_conquered_realm value = yes }
				# Story panel: increment event counter
				gr_story_increment_event = { TYPE = flag:war }
			}

			# === VARANGIAN DETECTION (Norse-specific adventure CBs) ===
			if = {
				limit = {
					exists = scope:war
					# AGOT does not define fp1_varangian_adventurer_conquest or
					# fp1_scandi_adventurer_conquest. Referencing them made
					# using_cb fail PostValidate at load. Branch left inert.
					always = no
				}
				
				scope:attacker = {
					# Get the war's target title
					if = {
						limit = { exists = scope:war }
						scope:war = {
							ordered_in_list = {
								list = target_titles
								order_by = tier
								max = 1
								save_scope_as = war_target_title
							}
						}
					}
					
					# First Varangian adventure
					if = {
						limit = { NOT = { has_variable = gr_memory_varangian } }
						set_variable = { name = gr_memory_varangian value = yes }
						set_variable = { name = gr_varangian_count value = 1 }
						if = {
							limit = { exists = scope:defender }
							set_variable = { name = gr_varangian_first_target value = scope:defender }
							set_variable = { name = gr_varangian_target value = scope:defender }
						}
						if = {
							limit = { exists = scope:war_target_title }
							set_variable = { name = gr_varangian_first_realm value = scope:war_target_title }
							set_variable = { name = gr_varangian_realm value = scope:war_target_title }
						}
					}
					# Subsequent Varangian adventures
					else = {
						change_variable = { name = gr_varangian_count add = 1 }
						if = {
							limit = { exists = scope:defender }
							set_variable = { name = gr_varangian_target value = scope:defender }
						}
						if = {
							limit = { exists = scope:war_target_title }
							set_variable = { name = gr_varangian_realm value = scope:war_target_title }
						}
					}
				}
			}
			
			# === GENERIC ADVENTURE DETECTION (landless adventurer CBs) ===
			else_if = {
				limit = {
					exists = scope:war
					OR = {
						scope:war = { using_cb = ep3_laamp_adventurer_duchy_invasion }
						scope:war = { using_cb = ep3_laamp_adventurer_kingdom_invasion }
						scope:war = { using_cb = ep3_laamp_adventurer_empire_invasion }
						scope:war = { using_cb = ep3_laamp_adventurer_county_invasion }
						scope:war = { using_cb = legendary_adventure }
					}
				}
				
				scope:attacker = {
					# Get the war's target title
					if = {
						limit = { exists = scope:war }
						scope:war = {
							ordered_in_list = {
								list = target_titles
								order_by = tier
								max = 1
								save_scope_as = war_target_title
							}
						}
					}
					
					# Save target title tier for flavor text (imperial throne vs throne etc)
					if = {
						limit = { exists = scope:war_target_title }
						if = {
							limit = { scope:war_target_title = { tier = tier_empire } }
							set_variable = { name = gr_adventure_tier value = flag:empire }
						}
						else_if = {
							limit = { scope:war_target_title = { tier = tier_kingdom } }
							set_variable = { name = gr_adventure_tier value = flag:kingdom }
						}
						else = {
							set_variable = { name = gr_adventure_tier value = flag:duchy }
						}
					}
					
					# First adventure
					if = {
						limit = { NOT = { has_variable = gr_memory_adventure } }
						set_variable = { name = gr_memory_adventure value = yes }
						set_variable = { name = gr_adventure_count value = 1 }
						if = {
							limit = { exists = scope:defender }
							set_variable = { name = gr_adventure_first_target value = scope:defender }
							set_variable = { name = gr_adventure_target value = scope:defender }
						}
						if = {
							limit = { exists = scope:war_target_title }
							set_variable = { name = gr_adventure_first_realm value = scope:war_target_title }
							set_variable = { name = gr_adventure_realm value = scope:war_target_title }
						}
					}
					# Subsequent adventures
					else = {
						change_variable = { name = gr_adventure_count add = 1 }
						if = {
							limit = { exists = scope:defender }
							set_variable = { name = gr_adventure_target value = scope:defender }
						}
						if = {
							limit = { exists = scope:war_target_title }
							set_variable = { name = gr_adventure_realm value = scope:war_target_title }
						}
					}

					# Story panel: increment event counter (adventure)
					gr_story_increment_event = { TYPE = flag:adventure }
				}
			}

			# === NOMAD INVASION DETECTION (1.19 Scribe - nomad CBs) ===
			else_if = {
				limit = {
					exists = scope:war
					OR = {
						scope:war = { using_cb = mpo_nomad_invasion_cb }
						scope:war = { using_cb = mpo_nomad_duchy_invasion_cb }
					}
				}

				scope:attacker = {
					# Get the war's target title
					if = {
						limit = { exists = scope:war }
						scope:war = {
							ordered_in_list = {
								list = target_titles
								order_by = tier
								max = 1
								save_scope_as = war_target_title
							}
						}
					}

					# First nomad invasion
					if = {
						limit = { NOT = { has_variable = gr_memory_nomad_invasion } }
						set_variable = { name = gr_memory_nomad_invasion value = yes }
						set_variable = { name = gr_nomad_invasion_count value = 1 }
						if = {
							limit = { exists = scope:defender }
							set_variable = { name = gr_nomad_invasion_first_target value = scope:defender }
							set_variable = { name = gr_nomad_invasion_target value = scope:defender }
						}
						if = {
							limit = { exists = scope:war_target_title }
							set_variable = { name = gr_nomad_invasion_first_realm value = scope:war_target_title }
							set_variable = { name = gr_nomad_invasion_realm value = scope:war_target_title }
						}
					}
					# Subsequent nomad invasions
					else = {
						change_variable = { name = gr_nomad_invasion_count add = 1 }
						if = {
							limit = { exists = scope:defender }
							set_variable = { name = gr_nomad_invasion_target value = scope:defender }
						}
						if = {
							limit = { exists = scope:war_target_title }
							set_variable = { name = gr_nomad_invasion_realm value = scope:war_target_title }
						}
					}
				}
			}

			# === CHINESE EXPANSION DETECTION (1.19 Scribe - China DLC CBs) ===
			else_if = {
				limit = {
					exists = scope:war
					# AGOT does not define the China DLC conquest CBs
					# (china_seize_county_cb, china_seize_duchy_cb,
					# chinese_consolidation_cb, grand_campaign_kingdom_invasion_cb).
					# Referencing them made using_cb fail PostValidate at load.
					# Branch left inert.
					always = no
				}

				scope:attacker = {
					# Get the war's target title
					if = {
						limit = { exists = scope:war }
						scope:war = {
							ordered_in_list = {
								list = target_titles
								order_by = tier
								max = 1
								save_scope_as = war_target_title
							}
						}
					}

					# First Chinese campaign
					if = {
						limit = { NOT = { has_variable = gr_memory_chinese_campaign } }
						set_variable = { name = gr_memory_chinese_campaign value = yes }
						set_variable = { name = gr_chinese_campaign_count value = 1 }

						}
					}
				}

		}
	}

# GOVERNMENT CHANGE TRACKING
# Root = character who changed government
# scope:old_government = old government type
# ================================================
on_government_change = {
	on_actions = {
		gr_internal_govt_change_tracking
	}
}

gr_internal_govt_change_tracking = {
	effect = {
		if = {
			limit = {
				gr_is_tracked_character = yes
				# Filter out government changes on succession
				NOT = { has_variable = gr_succession_lock }
			}

			set_variable = { name = gr_memory_govt_change value = yes }
			# Story panel: increment event counter
			gr_story_increment_event = { TYPE = flag:govt }
			
			# Detect old government type
			if = {
				limit = { scope:old_government = government_type:tribal_government }
				set_variable = { name = gr_govt_old value = flag:tribal }
			}
			else_if = {
				limit = { scope:old_government = government_type:feudal_government }
				set_variable = { name = gr_govt_old value = flag:feudal }
			}
			else_if = {
				limit = { scope:old_government = government_type:clan_government }
				set_variable = { name = gr_govt_old value = flag:clan }
			}
			else_if = {
				limit = { scope:old_government = government_type:administrative_government }
				set_variable = { name = gr_govt_old value = flag:admin }
			}
			else_if = {
				limit = { scope:old_government = government_type:nomad_government }
				set_variable = { name = gr_govt_old value = flag:nomad }
			}
			else = {
				set_variable = { name = gr_govt_old value = flag:other }
			}
			
			# Detect new government type (current)
			if = {
				limit = { government_has_flag = government_is_tribal }
				set_variable = { name = gr_govt_new value = flag:tribal }
			}
			else_if = {
				limit = { government_has_flag = government_is_feudal }
				set_variable = { name = gr_govt_new value = flag:feudal }
			}
			else_if = {
				limit = { government_has_flag = government_is_clan }
				set_variable = { name = gr_govt_new value = flag:clan }
			}
			else_if = {
				limit = { government_has_flag = government_is_administrative }
				set_variable = { name = gr_govt_new value = flag:admin }
			}
			else_if = {
				limit = { government_has_flag = government_is_nomadic }
				set_variable = { name = gr_govt_new value = flag:nomad }
			}
			else = {
				set_variable = { name = gr_govt_new value = flag:other }
			}
		}
	}
}

# FAITH CONVERSION TRACKING
# Root = the character who converted
# scope:old_faith = previous faith
# ================================================
on_character_faith_change = {
	on_actions = {
		gr_internal_faith_convert_tracking
	}
}

gr_internal_faith_convert_tracking = {
	effect = {
		if = {
			limit = {
				gr_is_tracked_character = yes
				# Landless adventurers are rulers too -- an is_landed-only
				# gate silently skipped faith-crossing tracking for the
				# whole adventurer playstyle (audit improvement #6).
				OR = {
					is_landed = yes
					is_ruler = yes
				}
				exists = scope:old_faith
			}
			
			# If faith creation was already tracked (on_faith_created fires first),
			# use this more reliable scope:old_faith to populate the old faith name
			if = {
				limit = { has_variable = gr_memory_faith_created }
				set_variable = { name = gr_faith_created_old value = scope:old_faith }
			}
			
			set_variable = { name = gr_memory_faith_convert value = yes }
			set_variable = { name = gr_faith_old value = scope:old_faith }
			# Story panel: increment event counter
			gr_story_increment_event = { TYPE = flag:faith }
		}
	}
}

# ================================================
# FAITH CREATED TRACKING (Reform or Divergent)
# Root = creator character
# scope:old_faith = the faith they had before
# ================================================
on_faith_created = {
	on_actions = {
		gr_internal_faith_created_tracking
	}
}

gr_internal_faith_created_tracking = {
	effect = {
		if = {
			limit = {
				gr_is_tracked_character = yes
				exists = scope:old_faith
			}
			
			set_variable = { name = gr_memory_faith_created value = yes }
			set_variable = { name = gr_faith_created_old value = scope:old_faith }
			
			# Was the old faith unreformed? Then this is a reformation
			if = {
				limit = {
					scope:old_faith = { has_doctrine_parameter = unreformed }
				}
				set_variable = { name = gr_faith_created_type value = flag:reformed }
			}
			# Otherwise it is a divergent/heretical faith
			else = {
				set_variable = { name = gr_faith_created_type value = flag:divergent }
			}
		}
	}
}

# ================================================
# CULTURE CREATED TRACKING (Hybrid or Divergent)
# Root = the NEW culture (not the character!)
# scope:founder = the founder character
# ================================================
on_culture_created = {
	on_actions = {
		gr_internal_culture_created_tracking
	}
}

gr_internal_culture_created_tracking = {
	effect = {
		if = {
			limit = {
				exists = scope:founder
				scope:founder = { gr_is_tracked_character = yes }
			}
			
			scope:founder = {
				set_variable = { name = gr_memory_culture_created value = yes }

				# Hybrid = two parent cultures, divergent = one
				if = {
					limit = {
						root = { is_hybrid_culture = yes }
					}
					set_variable = { name = gr_culture_created_type value = flag:hybrid }
				}
				else = {
					set_variable = { name = gr_culture_created_type value = flag:divergent }
				}
			}

			# Store parent cultures for localization.
			# Read them off ROOT (the new culture) rather than scope:parent_culture_1/2:
			# those two are saved inside vanilla's own on_culture_created effect block
			# and do not reach this chained on_action, so the old code silently
			# stored nothing and the obituary rendered "the traditions of  and  ".
			random_parent_culture = { save_temporary_scope_as = gr_parent_a }
			if = {
				limit = { any_parent_culture = { count >= 2 } }
				random_parent_culture = {
					limit = { this != scope:gr_parent_a }
					save_temporary_scope_as = gr_parent_b
				}
			}
			scope:founder = {
				if = {
					limit = { exists = scope:gr_parent_a }
					set_variable = { name = gr_culture_old value = scope:gr_parent_a }
				}
				if = {
					limit = { exists = scope:gr_parent_b }
					set_variable = { name = gr_culture_parent value = scope:gr_parent_b }
				}
			}
		}
	}
}

# ================================================
# BUILDING CONSTRUCTION TRACKING
# Tracks building upgrades/constructions within existing holdings.
# Simple counter with tiered obituary text.
# Chained via on_actions to avoid overriding vanilla's effect block.
# ================================================
on_building_completed = {
	on_actions = { gr_on_building_completed }
}
gr_on_building_completed = {
	effect = {
		# root = province, scope:character = the builder
		scope:character ?= {
			if = {
				limit = {
					gr_is_tracked_character = yes
					NOT = { has_variable = gr_succession_lock }
				}
				
				if = {
					limit = { NOT = { has_variable = gr_building_count } }
					set_variable = { name = gr_building_count value = 0 }
				}
				
				change_variable = { name = gr_building_count add = 1 }

				# M5: New holding detection.
				# same way (bare, on the province) in
				# common/buildings/00_standard_economy_buildings.txt:3917/3926
				# ("num_buildings < 5" / "num_buildings >= 8" inside a building's
				# ai_will_do, evaluated with root/this = the province). Here root is
				# already the province (per on_building_completed's own scope, see
				# section header above), so root's building count having reached
				# exactly 1 at this exact completion event means this was the
				# holding's FIRST building -- i.e. the holding was just founded.
				if = {
					limit = {
						root = { num_buildings = 1 }
					}
					if = {
						limit = { NOT = { has_variable = gr_holding_count } }
						set_variable = { name = gr_holding_count value = 0 }
					}
					change_variable = { name = gr_holding_count add = 1 }

					# Temple holding specifically.
					# syntax used in this exact scope (province root) in vanilla's own
					# on_building_completed handler, common/on_action/province_on_actions.txt:20:
					#   root = { if = { limit = { ... OR = {
					#       has_holding_type = church_holding
					#       has_holding_type = temple_citadel_holding
					#   } ... } } }
					# We count temple_citadel_holding too (upgraded temple holding),
					# matching vanilla's own OR grouping for "this is a temple holding".
					if = {
						limit = {
							root = {
								OR = {
									has_holding_type = church_holding
									has_holding_type = temple_citadel_holding
								}
							}
						}
						if = {
							limit = { NOT = { has_variable = gr_temple_count } }
							set_variable = { name = gr_temple_count value = 0 }
						}
						change_variable = { name = gr_temple_count add = 1 }
					}
				}

				# Track university buildings for learning score
				# Check if the province has a university building after completion
				# Uses a flag to prevent double-counting from same province
				if = {
					limit = {
						root = { has_building_with_flag = special_university }
						NOT = { has_variable_list = gr_university_provinces }
					}
					# First university ever — init list and count
					set_variable = { name = gr_university_built value = 1 }
					add_to_variable_list = { name = gr_university_provinces target = root.barony }
				}
				else_if = {
					limit = {
						root = { has_building_with_flag = special_university }
						has_variable_list = gr_university_provinces
						NOT = { is_target_in_variable_list = { name = gr_university_provinces target = root.barony } }
					}
					if = {
						limit = { NOT = { has_variable = gr_university_built } }
						set_variable = { name = gr_university_built value = 0 }
					}
					change_variable = { name = gr_university_built add = 1 }
					add_to_variable_list = { name = gr_university_provinces target = root.barony }
				}
			}
		}
	}
}

# ================================================
# ACTIVITY TRACKING
# ================================================
# There is NO global activity-completion on_action in vanilla, and the
# per-type on_start/on_complete/on_end blocks live INSIDE the activity_type
# database objects (common/activities/activity_types/*.txt) -- overriding
# those would clobber vanilla content, so we must not. Instead we additively
# hook one REAL, reusable named on_action per activity type: each activity
# type schedules these via "trigger_event = { on_action = X }" from its phase
# blocks (on_phase_active / on_start), and the on_action objects themselves
# are additive across files -- so extending them here is safe.
#
# SCOPE at fire time: for every hook below, ROOT/THIS is the character the
# on_action was triggered on and scope:activity exists (these are activity
# on_actions). We guard gr_is_tracked_character on that character.
#
# DEDUP (MANDATORY -- most of these hooks fire once per phase-pulse, not once
# per activity): per type we store the last-counted activity in a single
# self-overwriting var gr_last_act_<type> and only increment when the current
# scope:activity differs from it. The "?=" comparison is dangling-safe: after
# an activity ends its scope ref goes stale, and "var ?= scope" evaluates
# false rather than erroring (vanilla idiom, e.g.
# hunt_on_actions.txt:2196 "NOT = { var:sighting_owner ?= scope:host }").
# One var per type means no list growth.
#
# COVERAGE differs per hook (documented per-block): host-only hooks fire only
# for the activity host; all-participant hooks (wedding conclusion, pilgrimage
# arrival, university arrival) also count a tracked character who merely
# attended -- which is the honest reading ("attended a grand wedding").
#
# UNTRACKABLE via this route (no reusable named on_action exists in the
# activity_type -- they deliver events only via direct "trigger_event = <id>"
# in DB-object effect blocks we cannot extend): TOUR (activity_tour),
# WITCH_RITUAL/Grand Rite (activity_witch_ritual), GRUESOME_FESTIVAL/Blót
# (activity_gruesome_festival). Their counters (gr_tour_count, gr_ritual_count,
# gr_blot_count) are left unset -- their score/obituary branches simply stay
# dark rather than firing on a bad proxy.
# ================================================

# is the real vanilla hook fired on a successful murder. Its effect block opens
# with "scope:owner = { ... }" and branches on "scope:target" (the victim) right
# away, e.g.:
#   scope:owner = {
#       if = { limit = { scope:target = { is_incapable = yes } } ... }
#       ...
#   }

				if = {
					limit = { NOT = { has_variable = gr_succession_lock } }
					if = {
						limit = { NOT = { has_variable = gr_murder_count } }
						set_variable = { name = gr_murder_count value = 0 }

# Execution of a prisoner
# NOTE: on_execute_prisoner does NOT exist in vanilla CK3 -- there is no
# dedicated "execution" on_action. Executions are just a death with a
# specific death_reason, applied via the generic "death = { death_reason = ... }"
# effect from whichever decision/event carries it out. So we detect them off
# on_death instead (already additively extended by gr_on_actions.txt via its
# own on_actions = { ... } list -- we extend the SAME vanilla on_action here,
# additively, from this file, exactly like on_title_lost above).
# VERIFIED death_reason usage from vanilla events (grep across game/events/):
#   ep1_fund_inspiration_events.txt:21802
#     scope:inspiration_owner = { death = { death_reason = death_execution killer = root } }
#   bookmark_events.txt:266 (Blood Eagle option)
#     scope:prisoner = { death = { death_reason = death_execution_blood_eagle killer = root } }
#   court_events_general_1.txt:801-802
#     death = { death_reason = death_execution killer = scope:grander_ruler }
# In every real usage found, death_reason = death_execution or
# death_execution_blood_eagle is applied to the VICTIM (death fires in their
# scope) with killer = <executioner> set alongside -- matching on_death's own
# header comment: "if a killer is known, it's set as scope:killer". No vanilla
# event was found applying death_execution_killer, death_execution_blood_eagle_killer,
# or death_execution_blot via a death effect (death_execution_blot only shows up
# in an unrelated flavor trigger list in 00_death_management_triggers.txt), so
# those three are excluded here to avoid guessing at unused reason IDs.
on_death = {
	on_actions = {
		gr_internal_execution_tracking
	}
}

gr_internal_execution_tracking = {
	effect = {
		# root = the character who died (the victim); scope:killer = the executioner, if any
		if = {
			limit = {
				exists = scope:killer
				scope:killer = { gr_is_tracked_character = yes }
				OR = {
					death_reason = death_execution
					death_reason = death_execution_blood_eagle
				}
			}
			scope:killer = {
				if = {
					limit = { NOT = { has_variable = gr_succession_lock } }
					if = {
						limit = { NOT = { has_variable = gr_execution_count } }
						set_variable = { name = gr_execution_count value = 0 }
					}
					change_variable = { name = gr_execution_count add = 1 }
				}
			}
		}
	}

# ================================================
# TORTURE TRACKING (lossy-but-honest poll)
# ================================================
# There is NO torture on_action in vanilla. The torture_interaction
# (common/character_interactions/00_prison_interactions.txt:7307) fires
# events by id and leaves NO flag on the torturer. Only the VICTIM is marked:
# torturer_interaction_recipient_effect (common/scripted_effects/
# 00_interaction_effects.txt:361) does
#   scope:recipient = { add_character_modifier = { modifier = recently_tortured  years = 5 } }
# (modifier defined common/modifiers/00_event_modifiers.txt:210). So we cannot
# observe the act directly; we POLL for its footprint instead.
#
# Approach: on quarterly_playable_pulse (vanilla, yearly_on_actions.txt:2483 --
# "Root is the character"; already used by this mod in gr_story_on_actions.txt
# and gr_chronicle_migration_on_actions.txt), fast-gated is_ai = no to match
# the mod's other pulse usage, walk the pulsing character's own prisoners and
# count those currently carrying recently_tortured that we have not already
# counted (tracked in the variable-list gr_tortured_seen).
#
# HONEST CAVEATS (this is inference, not observation):
#  * ATTRIBUTION IS INFERRED. We credit the jailer (the character who holds the
#    prisoner). In principle another character with dungeon access could have
#    performed the torture; we cannot distinguish that, so the jailer is
#    credited.
#  * POLL GRANULARITY. Detection lags the act by up to one quarterly pulse
#    (~3 months). A torture whose 5-year modifier somehow lapses before the
#    next pulse would be missed, but 5y >> one quarter, so this is negligible.
#  * SAME-PRISONER RE-TORTURE within the 5-year modifier window counts ONCE
#    (the prisoner stays in gr_tortured_seen until the modifier expires). We
#    prune gr_tortured_seen of prisoners whose modifier has expired first, so a
#    prisoner tortured again years later (after expiry) is counted again.
quarterly_playable_pulse = {
	trigger = {
		is_ai = no
	}
	on_actions = {
		gr_internal_torture_poll
	}
}

gr_internal_torture_poll = {
	effect = {
		# root = the pulsing (player) character
		if = {
			limit = {
				gr_is_tracked_character = yes
				NOT = { has_variable = gr_succession_lock }
			}
			# 1) Prune gr_tortured_seen: drop anyone whose recently_tortured has
			#    expired, so a later re-torture (after 5y) is counted afresh.
			#    Removing from the same list we iterate is the vanilla prune
			#    idiom (e.g. common/scripted_effects/10_dlc_tgp_house_bloc_
			#    scripted_effects.txt:962-972).
			if = {
				limit = { has_variable_list = gr_tortured_seen }
				every_in_list = {
					variable = gr_tortured_seen
					limit = {
						NOT = { has_character_modifier = recently_tortured }
					}
					save_temporary_scope_as = gr_expired_prisoner
					root = {
						remove_list_variable = {
							name = gr_tortured_seen
							target = scope:gr_expired_prisoner
						}
					}
				}
			}
			# 2) Count each of my prisoners currently carrying recently_tortured
			#    that is not already recorded, and record them. We save the
			#    prisoner to a scope first (avoids fragile "prev" across the
			#    root-scope switch) and do the "already counted?" test on root
			#    with that explicit target.
			every_prisoner = {
				limit = {
					has_character_modifier = recently_tortured
				}
				save_temporary_scope_as = gr_new_torture_victim
				root = {
					if = {
						limit = {
							NOT = {
								is_target_in_variable_list = { name = gr_tortured_seen target = scope:gr_new_torture_victim }
							}
						}
						if = {
							limit = { NOT = { has_variable = gr_torture_count } }
							set_variable = { name = gr_torture_count value = 0 }
						}
						change_variable = { name = gr_torture_count add = 1 }
						add_to_variable_list = { name = gr_tortured_seen target = scope:gr_new_torture_victim }
					}
				}
			}
		}
	}
}
