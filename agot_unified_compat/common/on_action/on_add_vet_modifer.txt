on_combat_end_winner = {
    on_actions = {
        on_add_vet_modifier
    }
}

on_combat_end_loser = {
    on_actions = {
        on_add_vet_modifier
    }
}

on_add_vet_modifier = {
    effect = {
        every_side_knight = {
            if = {
                limit = {
                    is_ruler = no
                    # characters with no recorded father fail it, and every_side_knight runs it
                    # on both sides of every combat - 7,564 'father trigger [ Failed context
                    # switch ]' errors in one 37-minute session. '?=' makes it a soft switch:
                    # absent father simply fails the test instead of erroring.
                    # NOTE: 'is_army_owner' is not a valid CK3 trigger (the game logs
                    # "Unknown trigger: is_army_owner" at load), so this condition never
                    # evaluated meaningfully upstream either. Left as-is deliberately - fixing
                    # the spam is a bug fix, guessing the author's intent would be a rewrite.
                    NOT = {
                        father ?= { is_army_owner = root.army_owner }
                    }
                }
                add_character_modifier = veteran_modifier
            }
        }
    }
}
